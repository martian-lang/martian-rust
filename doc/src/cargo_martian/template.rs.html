158pub fn new_stage(
159    stage_name: impl AsRef<str>,
160    workspace_root: impl AsRef<Path>,
161    pkg: Option<impl AsRef<Path>>,
162    main_only: bool,
163) {
164    let stage_name = stage_name.as_ref();
165    let mut path = PathBuf::from(workspace_root.as_ref());
166    if let Some(p) = pkg {
167        path.push(p);
168    }
169    path.push("src");
170    path.push(stage_name.to_snake_case());
171    path.set_extension("rs");
172
173    // Make sure the file does not exist already
174    assert!(!path.exists(), "File {path:?} already exists");
175
176    let vars = [
177        ("stage".to_string(), stage_name.to_upper_camel_case()),
178        ("open".to_string(), "{".to_string()),
179        ("close".to_string(), "}".to_string()),
180    ]
181    .into_iter()
182    .collect();
183
184    let stage_template = if main_only {
185        strfmt(STAGE_TEMPLATE_MAIN, &vars).unwrap()
186    } else {
187        strfmt(STAGE_TEMPLATE, &vars).unwrap()
188    };
189
190    println!("Writing to file {path:?}");
191    let mut f = File::create(path).expect("Failed to create file");
192    write!(f, "{stage_template}").expect("Failed writing to file");
193}

270pub fn new_adapter(adapter_name: impl AsRef<str>) {
271    let adapter_name = adapter_name.as_ref().to_snake_case();
272    let exit_status = std::process::Command::new("cargo")
273        .arg("new")
274        .arg(&adapter_name)
275        .spawn()
276        .expect("could not run cargo new")
277        .wait()
278        .expect("failed to wait for cargo new?");
279
280    if exit_status.success() {
281        {
282            // Main file
283            let vars = [
284                ("adapter".to_string(), adapter_name.to_snake_case()),
285                ("open".to_string(), "{".to_string()),
286                ("close".to_string(), "}".to_string()),
287            ]
288            .into_iter()
289            .collect();
290
291            let mut path = PathBuf::from(&adapter_name);
292            path.push("src");
293            path.push("main.rs");
294
295            {
296                // Play it safe
297                assert!(path.exists());
298                let mut f = File::open(&path).expect("Couldn't open main.rs for writing");
299                let mut contents = String::new();
300                f.read_to_string(&mut contents)
301                    .expect("Could not read the main.rs file");
302                assert!(contents == DEFAULT_MAIN);
303            }
304
305            let main_template = strfmt(ADAPTER_MAIN_TEMPLATE, &vars).unwrap();
306            eprintln!("Writing main template to {path:?}");
307            let mut f = File::create(path).expect("Failed to create file");
308            write!(f, "{main_template}").expect("Failed writing to main.rs file");
309        }
310
311        {
312            // Cargo toml
313            let mut path = PathBuf::from(&adapter_name);
314            path.push("Cargo.toml");
315
316            let mut f = OpenOptions::new()
317                .append(true)
318                .open(path)
319                .expect("Couldn't open Cargo.toml for writing");
320            write!(f, "{CARGO_TOML_ADDITION}").expect("Failed writing to Cargo.toml file");
321        }
322    }
