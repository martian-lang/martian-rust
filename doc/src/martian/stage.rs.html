24fn split_file_name(p: &Path) -> (&Path, &Path) {
25    let file_name = p.file_name().unwrap();
26    match p.parent() {
27        Some(path) => (path, file_name.as_ref()),
28        None => ("".as_ref(), file_name.as_ref()),
29    }
30}

32/// A `MartianFiletype` is associated with a file of know non-empty
33/// extension. This encodes the concept of a `filepath` in martian.
34pub trait MartianFileType: AsRef<Path> + From<PathBuf> {
35    fn extension() -> String;
36    fn new(file_path: impl AsRef<Path>, file_name: impl AsRef<Path>) -> Self;
37    /// This function is equivalent to calling `new(p.parent(), p.file_name())`
38    /// except that it handles the case of the path lacking one or the other.
39    fn from_path(p: &Path) -> Self {
40        let (path, file_name) = split_file_name(p);
41        Self::new(path, file_name)
42    }
43    /// This function will create a file if it does not exist, and will truncate it if it does.
44    fn buf_writer(&self) -> Result<BufWriter<File>, Error> {
45        fn _buf_writer(ty: &Path) -> Result<BufWriter<File>, Error> {
46            Ok(BufWriter::new(File::create(ty).map_err(|e| {
47                let context = format!(
48                    "Failed to create file '{}' from within MartianType::buf_writer() due to {:?}",
49                    ty.display(),
50                    e
51                );
52                Error::new(e).context(context)
53            })?))
54        }
55        _buf_writer(self.as_ref())
56    }
57    fn buf_reader(&self) -> Result<BufReader<File>, Error> {
58        fn _buf_reader(ty: &Path) -> Result<BufReader<File>, Error> {
59            Ok(BufReader::new(File::open(ty).map_err(|e| {
60                let context = format!(
61                    "Failed to open file '{}' from within MartianType::buf_reader() due to {:?}",
62                    ty.display(),
63                    e
64                );
65                Error::new(e).context(context)
66            })?))
67        }
68        _buf_reader(self.as_ref())
69    }
70}

303/// All the chunks in the stage (with their inputs & resource)
304/// along with the join resource. This needs to be constructed
305/// in the `split()` function, so that martian can create chunks
306/// appropriately and set resource reservations for `main()` and
307/// `join()`
308///
309/// Take a look at the `split()` function [here for a concrete example.](https://github.com/martian-lang/martian-rust/blob/master/martian-lab/examples/sum_sq/src/sum_squares.rs#L61)
310///
311/// `StageDef` is generic over type `T` which is the type of `ChunkInputs`
312///
313/// the toy example below constructs 100 chunks.
314/// ```rust
315/// use martian::prelude::*;
316///
317/// struct ChunkInputs {
318///     chunk_id: usize,
319/// }
320///
321/// let join_resource = Resource::new().mem_gb(16).threads(4); //
322/// let mut stage_def = StageDef::with_join_resource(join_resource);
323/// // You can use StageDef::new() for default join resource allocation
324///
325/// for chunk_id in 0..100 {
326///    let chunk_inputs = ChunkInputs { chunk_id };
327///    let chunk_resource = Resource::with_mem_gb(3); // Could be a function of chunk inputs
328///    stage_def.add_chunk_with_resource(chunk_inputs, chunk_resource);
329///    // You can use stage_def.add_chunk(chunk_inputs) for default chunk resource allocation
330/// }
331/// ```
332#[derive(Debug, Default, Serialize, Deserialize)]
333pub struct StageDef<T> {
334    chunks: Vec<ChunkDef<T>>,
335    #[serde(rename = "join")]
336    join_resource: Resource,
337}
338

421/// martian run as well as the actual resources available for you.

432impl From<SharedMetadata> for MartianRover {
433    fn from(smd: SharedMetadata) -> MartianRover {
434        let md = smd.lock().unwrap();
435        let files_path = PathBuf::from(&md.files_path);
436        let mem_gb = md.jobinfo.mem_gb;
437        let threads = md.jobinfo.threads;
438        let vmem_gb = md.jobinfo.vmem_gb;
439        let version = md.jobinfo.version.clone();
440        drop(md);
441        MartianRover {
442            files_path,
443            mem_gb,
444            threads,
445            vmem_gb,
446            version,
447            metadata: Some(smd),
448        }
449    }

539/// Two different kinds of marian stages. `MainOnly` stages only have a
540/// `main()` function whereas `WithSplit` stages have `split()` and `join()` too.
541///
542/// `MartianStage` trait has a default implementation for the function `stage_kind()`
543/// which returns `StageKind::WithSplit`. `MartianMain` trait overrides this to return
544/// `StageKind::MainOnly`. This enum exists so that one can determine whether a stage object
545/// implements `MartianMain` or `MartianStage`
546#[derive(Debug)]
547pub enum StageKind {
548    /// Stage with only a `main()` function
549    MainOnly,
550    /// Stage with `split()`, `main()` and `join()` functions
551    WithSplit,
552}

554/// A stage in martian with just the main function.
555///
556/// For a toy example, see: [https://martian-lang.github.io/martian-rust/#/content/quick_start](https://martian-lang.github.io/martian-rust/#/content/quick_start)
557pub trait MartianMain: MroMaker {
558    type StageInputs: DeserializeOwned + MartianStruct;
559    type StageOutputs: Serialize + DeserializeOwned + MartianStruct;
560
561    fn main(
562        &self,
563        args: Self::StageInputs,
564        rover: MartianRover,
565    ) -> Result<Self::StageOutputs, Error>;
566}

568fn split_prelude(
569    run_directory: &Path,
570    stage_name: &str,
571    subdir: &str,
572) -> Result<MartianRover, Error> {
573    // Use default resource for split
574    let default_resource = Resource::new().mem_gb(1).vmem_gb(2).threads(1);
575    let split_path = prep_path(run_directory, subdir)?;
576    let rover = MartianRover::new(split_path.as_path(), default_resource);
577    println!("{}", ["-"; 80].concat());
578    println!("{stage_name}");
579    println!("{}", ["-"; 80].concat());
580    Ok(rover)

583/// A stage in martian which has `split`, `main` and `join`
584///
585/// For a toy example, see [https://martian-lang.github.io/martian-rust/#/content/quick_start_split](https://martian-lang.github.io/martian-rust/#/content/quick_start_split)
586pub trait MartianStage: MroMaker {
587    type StageInputs: DeserializeOwned + MartianStruct;
588    type StageOutputs: Serialize + MartianStruct;
589    type ChunkInputs: Serialize + DeserializeOwned + MartianStruct;
590    type ChunkOutputs: Serialize + DeserializeOwned + MartianStruct;
591
592    fn split(
593        &self,
594        args: Self::StageInputs,
595        rover: MartianRover,
596    ) -> Result<StageDef<Self::ChunkInputs>, Error>;
597
598    fn main(
599        &self,
600        args: Self::StageInputs,
601        chunk_args: Self::ChunkInputs,
602        rover: MartianRover,
603    ) -> Result<Self::ChunkOutputs, Error>;
604
605    fn join(
606        &self,
607        args: Self::StageInputs,
608        chunk_defs: Vec<Self::ChunkInputs>,
609        chunk_outs: Vec<Self::ChunkOutputs>,
610        rover: MartianRover,
611    ) -> Result<Self::StageOutputs, Error>;
612
613    /// In-process stage runner, useful for writing unit tests that exercise one of more stages purely from Rust.
614    /// Executes stage with arguments `args` in directory `run_directory`. The defaul implementation executes split
615    /// to get the stage definition (chunks), executes each chunk one after another and finally calls the join function.
616    fn test_run(
617        &self,
618        run_directory: impl AsRef<Path> + Send + Sync,
619        args: Self::StageInputs,
620    ) -> Result<Self::StageOutputs, Error>
621    where
622        Self: Sync,
623        Self::ChunkInputs: Clone + Send + Sync,
624        Self::StageInputs: Clone + Send + Sync,
625        Self::ChunkOutputs: Send + Sync,
626    {
627        let run_directory = run_directory.as_ref();
628
629        let rover = split_prelude(run_directory, Self::stage_name(), "split")?;
630        println!(" > [split ] running");
631        let stage_defs = self.split(args.clone(), rover)?;
632        println!(" > [split ] complete");
633
634        let run_chunk = |chunk: &ChunkDef<Self::ChunkInputs>,
635                         chunk_idx: usize|
636         -> Result<Self::ChunkOutputs, Error> {
637            fn _chunk_prelude(
638                chunk_idx: usize,
639                run_directory: &Path,
640                resource: Resource,
641            ) -> Result<MartianRover, Error> {
642                #[cfg(not(feature = "rayon"))]
643                println!(" > [chunk ] running {}", chunk_idx,);
644                #[cfg(feature = "rayon")]
645                println!(" > [chunk ] running with rayon {chunk_idx}",);
646                let chunk_path = prep_path(run_directory, &format!("chnk{chunk_idx}"))?;
647                Ok(MartianRover::new(
648                    chunk_path.as_path(),
649                    fill_defaults(resource),
650                ))
651            }
652            let rover = _chunk_prelude(chunk_idx, run_directory, chunk.resource)?;
653            self.main(args.clone(), chunk.inputs.clone(), rover)
654        };
655
656        println!(" > [chunks] {} chunks in total", stage_defs.chunks.len());
657
658        #[cfg(not(feature = "rayon"))]
659        let chunk_outs = stage_defs
660            .chunks
661            .iter()
662            .enumerate()
663            .map(|(chunk_idx, chunk)| run_chunk(chunk, chunk_idx))
664            .collect::<Result<Vec<_>, Error>>()?;
665
666        #[cfg(feature = "rayon")]
667        let chunk_outs = stage_defs
668            .chunks
669            .par_iter()
670            .enumerate()
671            .map(|(chunk_idx, chunk)| run_chunk(chunk, chunk_idx))
672            .collect::<Result<Vec<_>, Error>>()?;
673
674        println!(" > [chunks] complete");
675
676        let join_path = prep_path(run_directory, "join")?;
677        let rover = MartianRover::new(join_path.as_path(), fill_defaults(stage_defs.join_resource));
678
679        let mut chunk_defs = Vec::new();
680        for c in stage_defs.chunks {
681            chunk_defs.push(c.inputs);
682        }
683
684        println!(" > [join  ] running");
685        let result = self.join(args, chunk_defs, chunk_outs, rover);
686        println!(" > [stage ] complete");
687        result
688    }

690    /// In-process stage runner, useful for writing unit tests that exercise one of more stages purely from Rust.
691    /// Executes stage with arguments `args` in temporary directory that will always be cleaned up.
692    fn test_run_tmpdir(&self, args: Self::StageInputs) -> Result<Self::StageOutputs, Error>
693    where
694        Self: Sync,
695        Self::ChunkInputs: Clone + Send + Sync,
696        Self::StageInputs: Clone + Send + Sync,
697        Self::ChunkOutputs: Send + Sync,
698    {
699        let tmp_dir = tempfile::tempdir()?;
700        self.test_run(&tmp_dir, args)
701    }
702    fn stage_kind() -> StageKind {
703        StageKind::WithSplit
704    }

707/// A raw martian stage that works with untype metadata. It is recommended
708/// not to implement this directly. Use `MartianMain` or `MartianStage` traits instead
709pub trait RawMartianStage {
710    fn split(&self, metadata: SharedMetadata) -> Result<(), Error>;
711    fn main(&self, metadata: SharedMetadata) -> Result<(), Error>;
712    fn join(&self, metadata: SharedMetadata) -> Result<(), Error>;
713}
714
715impl<T> MartianStage for T
716where
717    T: MartianMain,
718{
719    type StageInputs = <T as MartianMain>::StageInputs;
720    type StageOutputs = <T as MartianMain>::StageOutputs;
721    type ChunkInputs = MartianVoid;
722    type ChunkOutputs = <T as MartianMain>::StageOutputs;
723
724    fn split(&self, _: Self::StageInputs, _: MartianRover) -> Result<StageDef<MartianVoid>, Error> {
725        unimplemented!()
726    }
727
728    fn main(
729        &self,
730        args: Self::StageInputs,
731        _: MartianVoid,
732        rover: MartianRover,
733    ) -> Result<Self::ChunkOutputs, Error> {
734        <T as MartianMain>::main(self, args, rover)
735    }
736
737    fn join(
738        &self,
739        _: Self::StageInputs,
740        _: Vec<MartianVoid>,
741        _: Vec<Self::ChunkOutputs>,
742        _: MartianRover,
743    ) -> Result<Self::StageOutputs, Error> {
744        unimplemented!()
745    }
746
747    fn test_run(
748        &self,
749        run_directory: impl AsRef<Path>,
750        args: Self::StageInputs,
751    ) -> Result<Self::StageOutputs, Error> {
752        // Use default resource for main
753        let rover = split_prelude(run_directory.as_ref(), Self::stage_name(), "main")?;
754        println!(" > [chunk] running");
755        let result = self.main(args, rover);
756        println!(" > [stage] complete");
757        result
758    }
759
760    fn test_run_tmpdir(&self, args: Self::StageInputs) -> Result<Self::StageOutputs, Error>
761    where
762        Self: Sync,
763        Self::ChunkInputs: Clone + Send + Sync,
764        Self::StageInputs: Clone + Send + Sync,
765        Self::ChunkOutputs: Send + Sync,
766    {
767        let tmp_dir = tempfile::tempdir()?;
768        self.test_run(&tmp_dir, args)
769    }
770    fn stage_kind() -> StageKind {
771        StageKind::MainOnly
772    }
773}

775const ARGS_FN: &str = "args";
776const OUTS_FN: &str = "outs";
777
778impl<T> RawMartianStage for T
779where
780    T: MartianStage,
781{
782    fn split(&self, md: SharedMetadata) -> Result<(), Error> {
783        let args: <T as MartianStage>::StageInputs = md.lock().unwrap().decode(ARGS_FN)?;
784        let stage_defs = MartianStage::split(self, args, md.clone().into())?;
785        let stage_def_obj = obj_encode(&stage_defs)?;
786        md.lock()
787            .unwrap()
788            .complete_with("stage_defs", &stage_def_obj)
789    }
790
791    fn main(&self, md: SharedMetadata) -> Result<(), Error> {
792        let args: <T as MartianStage>::StageInputs = md.lock().unwrap().decode(ARGS_FN)?;
793        let chunk_args: <T as MartianStage>::ChunkInputs = md.lock().unwrap().decode(ARGS_FN)?;
794        let outs = MartianStage::main(self, args, chunk_args, md.clone().into())?;
795        let outs_obj = obj_encode(&outs)?;
796        md.lock().unwrap().complete_with(OUTS_FN, &outs_obj)
797    }
798
799    fn join(&self, md: SharedMetadata) -> Result<(), Error> {
800        let args: <T as MartianStage>::StageInputs = md.lock().unwrap().decode(ARGS_FN)?;
801        let chunk_defs: Vec<<T as MartianStage>::ChunkInputs> =
802            md.lock().unwrap().decode("chunk_defs")?;
803        let chunk_outs: Vec<<T as MartianStage>::ChunkOutputs> =
804            md.lock().unwrap().decode("chunk_outs")?;
805        let outs = MartianStage::join(self, args, chunk_defs, chunk_outs, md.clone().into())?;
806        let outs_obj = obj_encode(&outs)?;
807        md.lock().unwrap().complete_with(OUTS_FN, &outs_obj)
808    }
809}

811// Prep a path for a test run of a stage.
812fn prep_path(path: impl AsRef<Path>, subdir: &str) -> Result<PathBuf, Error> {
813    _prep_path(path.as_ref(), subdir)
814}
815
816fn _prep_path(path: &Path, subdir: &str) -> Result<PathBuf, Error> {
817    let sub_path = path.join(subdir);
818
819    std::fs::create_dir(&sub_path)?;
820    Ok(sub_path)
821}

823fn fill_defaults(mut resource: Resource) -> Resource {
824    if resource.mem_gb.is_none() {
825        resource.mem_gb.replace(1.);
826    }
827
828    if resource.vmem_gb.is_none() {
829        resource.vmem_gb.replace(2.);
830    }
831
832    if resource.threads.is_none() {
833        resource.threads.replace(1);
834    }
835
836    resource
837}
