105pub fn make_timestamp(datetime: impl Into<OffsetDateTime>) -> String {
106    _make_timestamp(datetime.into())

109fn _make_timestamp(datetime: OffsetDateTime) -> String {
110    // Convert to local time (if necessary, and possible)
111    let datetime = if let Ok(offset) = UtcOffset::local_offset_at(datetime) {
112        datetime.to_offset(offset)
113    } else {
114        datetime
115    };
116    datetime.format(DATE_FORMAT).unwrap()
117}

119pub fn make_timestamp_now() -> String {
120    make_timestamp(SystemTime::now())

123fn ignore_not_found(err: std::io::Error) -> std::io::Result<()> {
124    match err.kind() {
125        ErrorKind::NotFound => {
126            // Workaround for an issue on heavily loaded NFS servers.
127            // If a request is taking a long time, the client will
128            // re-send the request.  The server is supposed to note
129            // that the request is a duplicate and de-duplicate it,
130            // but if it's heavily loaded its duplicate request
131            // cache might have already been flushed, in which case
132            // the second request will see ENOENT and fail.
133            // There's no way to tell the difference between that
134            // not-actually-failure mode and something else deleting
135            // the file before rename, because after a successful
136            // rename it's expected that `mrp` may delete the file
137            // immediately.  So we have to just ignore this error.
138            Ok(())
139        }
140        _ => Err(err),
141    }
142}

173    ///
174    /// This operation is not atomic at the file system level.
175    pub(crate) fn append_metadata(&mut self, name: &str, content: &str) -> Result<()> {
176        let mut f = OpenOptions::new()
177            .create(true)
178            .append(true)
179            .open(self.make_path(name))?;
180        f.write_all(content.as_bytes())?;
181        f.sync_all()?;
182        drop(f);
183        self.update_journal(name)
184    }

190        fully_atomic_write(&self.make_path(name), content)?;

194    /// Update the Martian journal -- so that Martian knows what we've updated
195    fn update_journal(&self, name: &str) -> Result<()> {
196        let journal_filename = if self.stage_type == StageType::Main {
197            format!("{}.{name}", self.run_file)
198        } else {
199            format!("{}.{}_{name}", self.run_file, self.stage_type)
200        };
201        fully_atomic_write(Path::new(&journal_filename), &make_timestamp_now())
202    }

213    fn _decode<T: Sized + DeserializeOwned>(file: PathBuf) -> Result<T> {
214        let buf = Self::_read_buf_err(&file)?;
215        serde_json::from_str(&buf).map_err(
216            #[cold]
217            |e| Self::_format_buf_err(buf, e, file, type_name::<T>()),
218        )
219    }
220
221    fn _read_buf_err(file: &Path) -> Result<String> {
222        std::fs::read_to_string(file).map_err(
223            #[cold]
224            |e| {
225                let context = format!("Failed to read file {file:?} due to {e}:");
226                Error::new(e).context(context)
227            },
228        )
229    }
230
231    #[cold]
232    fn _format_buf_err(
233        buf: String,
234        e: serde_json::Error,
235        file: PathBuf,
236        tname: &'static str,
237    ) -> Error {
238        // Non-generic so that we don't generate copy of this code for every
239        // type we `_decode` into.  This is a slight hack to improve compile
240        // times.
241        let buf_lines: Vec<_> = buf
242            .lines()
243            .enumerate()
244            .map(|(i, line)| format!("{:>4}: {line}", i + 1))
245            .collect();
246        let context = format!(
247            "The martian-rust adapter failed while deserializing the file {:?} as {} due to the \
248            following error:\n\n{}\n\nThis typically happens when one or more fields in the \
249            struct {} cannot be built from the JSON. The contents of the JSON are shown below: \
250            \n{}",
251            file.file_name().unwrap(),
252            tname,
253            e,
254            tname,
255            buf_lines.join("\n")
256        );
257        Error::new(e).context(context)
258    }

276    pub fn update_jobinfo(&mut self) -> Result<()> {
277        let mut raw_jobinfo: JsonDict = self.decode("jobinfo")?;
278        let jobinfo: JobInfo = serde_json::from_value(Value::Object(raw_jobinfo.clone()))?;
279
280        let info = RustAdapterInfo::new();
281        raw_jobinfo.insert("rust".to_string(), serde_json::to_value(info)?);
282
283        self.write_json_obj("jobinfo", &raw_jobinfo)?;
284        self.raw_jobinfo = raw_jobinfo;
285        self.jobinfo = jobinfo;
286        Ok(())
287    }

333/// Write a new file atomically, by first writing to a temp file, fsync, then rename.
334///
335/// The temp version will be the provided path with .tmp appended.
336fn fully_atomic_write(path: &Path, content: &str) -> Result<()> {
337    let mut tmp_path = OsString::from(path);
338    tmp_path.push(".tmp");
339    let mut f = File::create(&tmp_path).with_context(|| tmp_path.display().to_string())?;
340    let write_result = f
341        .write_all(content.as_bytes())
342        .and_then(|_| f.sync_all())
343        .with_context(|| tmp_path.display().to_string());
344    drop(f);
345    if let Err(err) = write_result {
346        let _ = remove_file(&tmp_path);
347        return Err(err);
348    }
349    rename(&tmp_path, path).or_else(ignore_not_found)?;
350    Ok(())
351}
