107    _make_timestamp(datetime.into())

110fn _make_timestamp(datetime: OffsetDateTime) -> String {
111    // Convert to local time (if necessary, and possible)
112    let datetime = if let Ok(offset) = UtcOffset::local_offset_at(datetime) {
113        datetime.to_offset(offset)
114    } else {
115        datetime
116    };
117    datetime.format(DATE_FORMAT).unwrap()
118}

120pub fn make_timestamp_now() -> String {
121    make_timestamp(SystemTime::now())

124fn ignore_not_found(err: std::io::Error) -> std::io::Result<()> {
125    match err.kind() {
126        ErrorKind::NotFound => {
127            // Workaround for an issue on heavily loaded NFS servers.
128            // If a request is taking a long time, the client will
129            // re-send the request.  The server is supposed to note
130            // that the request is a duplicate and de-duplicate it,
131            // but if it's heavily loaded its duplicate request
132            // cache might have already been flushed, in which case
133            // the second request will see ENOENT and fail.
134            // There's no way to tell the difference between that
135            // not-actually-failure mode and something else deleting
136            // the file before rename, because after a successful
137            // rename it's expected that `mrp` may delete the file
138            // immediately.  So we have to just ignore this error.
139            Ok(())
140        }
141        _ => Err(err),
142    }
143}

173    /// Append to a metadata file inside the chunk.
174    ///
175    /// This operation is not atomic at the file system level.
176    pub(crate) fn append_metadata(&mut self, name: &str, content: &str) -> Result<()> {
177        let mut f = OpenOptions::new()
178            .create(true)
179            .append(true)
180            .open(self.make_path(name))?;
181        f.write_all(content.as_bytes())?;
182        f.sync_all()?;
183        drop(f);
184        self.update_journal(name)
185    }

187    /// Write to a metadata file, overwriting existing contents.
188    ///
189    /// The file write is fully atomic.
190    pub(crate) fn write_metadata(&mut self, name: &str, content: &str) -> Result<()> {
191        fully_atomic_write(&self.make_path(name), |w| w.write_all(content.as_bytes()))?;
192        self.update_journal(name)?;
193        Ok(())
194    }

196    /// Update the Martian journal -- so that Martian knows what we've updated
197    fn update_journal(&self, name: &str) -> Result<()> {
198        let journal_name: Cow<'_, str> = if self.stage_type != StageType::Main {
199            format!("{}_{name}", self.stage_type).into()
200        } else {
201            name.into()
202        };
203        let journal_file = format!("{}.{journal_name}", self.run_file);
204        let timestamp = make_timestamp_now();
205        let mut write_err = Ok(());
206        let create_result = fully_atomic_write(Path::new(&journal_file), |w| {
207            write_err = w.write_all(timestamp.as_bytes());
208            Ok(())
209        });
210        if let Err(err) = write_err {
211            // Pretty much ignore this error.  The only reason we need
212            // any content at all in this file is because some
213            // filesystems behave strangely with completely empty files.
214            eprintln!("Writing journal file {journal_file}.tmp: {err}");
215        }
216        create_result?;
217        Ok(())
218    }

229    fn _decode<T: Sized + DeserializeOwned>(file: PathBuf) -> Result<T> {
230        let buf = Self::_read_buf_err(&file)?;
231        serde_json::from_str(&buf).map_err(
232            #[cold]
233            |e| Self::_format_buf_err(buf, e, file, type_name::<T>()),
234        )
235    }
236
237    fn _read_buf_err(file: &Path) -> Result<String> {
238        std::fs::read_to_string(file).map_err(
239            #[cold]
240            |e| {
241                let context = format!("Failed to read file {file:?} due to {e}:");
242                Error::new(e).context(context)
243            },
244        )
245    }
246
247    #[cold]
248    fn _format_buf_err(
249        buf: String,
250        e: serde_json::Error,
251        file: PathBuf,
252        tname: &'static str,
253    ) -> Error {
254        // Non-generic so that we don't generate copy of this code for every
255        // type we `_decode` into.  This is a slight hack to improve compile
256        // times.
257        let buf_lines: Vec<_> = buf
258            .lines()
259            .enumerate()
260            .map(|(i, line)| format!("{:>4}: {line}", i + 1))
261            .collect();
262        let context = format!(
263            "The martian-rust adapter failed while deserializing the file {:?} as {} due to the \
264            following error:\n\n{}\n\nThis typically happens when one or more fields in the \
265            struct {} cannot be built from the JSON. The contents of the JSON are shown below: \
266            \n{}",
267            file.file_name().unwrap(),
268            tname,
269            e,
270            tname,
271            buf_lines.join("\n")
272        );
273        Error::new(e).context(context)
274    }

349/// Write a new file atomically, by first writing to a temp file, fsync, then rename.
350///
351/// The temp version will be the provided path with .tmp appended.
352fn fully_atomic_write(
353    path: &Path,
354    write: impl FnOnce(&mut dyn Write) -> std::io::Result<()>,
355) -> std::io::Result<()> {
356    let mut tmp_path: OsString = path.into();
357    tmp_path.push(".tmp");
358    let mut f = File::create(&tmp_path)?;
359    let temp_write_result = write(&mut f).and_then(|_| f.sync_all());
360    drop(f);
361    if let Err(err) = temp_write_result {
362        let _ = remove_file(&tmp_path);
363        return Err(err);
364    }
365
366    rename(&tmp_path, path).or_else(ignore_not_found)?;
367    Ok(())
368}
