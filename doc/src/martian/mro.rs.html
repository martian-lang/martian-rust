30/// Defines how an entity that denotes some part of the mro is displayed
31pub trait MroDisplay: Display {
32    fn mro_string(&self, field_width: Option<usize>) -> String {
33        match field_width {
34            Some(width) => {
35                let min_width = self.min_width();
36                assert!(
37                    width >= min_width,
38                    "Need a minimum width of {}. Found {}",
39                    min_width,
40                    width
41                );
42                self.mro_string_with_width(width)
43            }
44            None => self.mro_string_no_width(),
45        }
46    }
47    fn min_width(&self) -> usize;
48    fn mro_string_no_width(&self) -> String {
49        self.to_string()
50    }
51    fn mro_string_with_width(&self, field_width: usize) -> String {
52        format!("{self:<field_width$}")
53    }
54}

83impl Display for StructDef {
84    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
85        // Determing the field width for the type field
86        let ty_width = self
87            .fields
88            .iter()
89            .map(MroDisplay::min_width)
90            .max()
91            .unwrap_or(0);
92
93        // Determing the field width for the name field
94        let name_width = self
95            .fields
96            .iter()
97            .map(MroField::name_width)
98            .max()
99            .unwrap_or(0);
100
101        // Determing the field width for the desc field
102        let desc_width = self
103            .fields
104            .iter()
105            .map(MroField::desc_width)
106            .max()
107            .unwrap_or(0);
108
109        writeln!(f, "struct {}(", self.name)?;
110
111        for field in &self.fields {
112            let formatted_line = field.fmt_align_4_columns(ty_width, name_width, desc_width);
113            writeln!(f, "    {},", formatted_line)?;
114        }
115        writeln!(f, ")")
116    }
117}

183#[derive(Debug, Serialize, Deserialize, Clone, PartialEq, Eq, Hash)]
184pub enum MartianBlanketType {
185    Primary(MartianPrimaryType),
186    Array(Box<MartianBlanketType>),
187    TypedMap(Box<MartianBlanketType>),
188}

190impl MartianBlanketType {
191    fn inner(&self) -> MartianPrimaryType {
192        match self {
193            MartianBlanketType::Primary(ref primary) => primary.clone(),
194            MartianBlanketType::Array(ref blanket) => blanket.inner(),
195            MartianBlanketType::TypedMap(ref blanket) => blanket.inner(),
196        }
197    }
198}

200impl MroDisplay for MartianBlanketType {
201    fn min_width(&self) -> usize {
202        match *self {
203            MartianBlanketType::Primary(ref primary) => primary.min_width(),
204            MartianBlanketType::Array(ref blanket) => blanket.min_width() + 2,
205            MartianBlanketType::TypedMap(ref blanket) => {
206                // map of maps not allowed in Martian
207                // this is a little hacky, we allow TypedMap<map> to be passed around internally in Martian-rust
208                // but we just print it as "map"
209                match **blanket {
210                    MartianBlanketType::TypedMap(_)
211                    | MartianBlanketType::Primary(MartianPrimaryType::Map) => 3,
212                    // map<T>
213                    _ => blanket.min_width() + 5,
214                }
215            }
216        }
217    }
218}

220impl Display for MartianBlanketType {
221    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
222        match *self {
223            MartianBlanketType::Primary(ref primary) => {
224                write!(f, "{primary:<w$}", w = f.width().unwrap_or_default())
225            }
226            MartianBlanketType::Array(ref blanket) => {
227                write!(
228                    f,
229                    "{v}{a:<w$}",
230                    v = blanket,
231                    a = "[]",
232                    w = f
233                        .width()
234                        .unwrap_or_default()
235                        .saturating_sub(blanket.min_width())
236                )
237            }
238            MartianBlanketType::TypedMap(ref blanket) => {
239                // map of maps not allowed in Martian
240                // this is a little hacky, we allow TypedMap<map> to be passed around internally in Martian-rust
241                // but we just print it as "map"
242                match **blanket {
243                    MartianBlanketType::TypedMap(_)
244                    | MartianBlanketType::Primary(MartianPrimaryType::Map) => {
245                        write!(f, "{v:<w$}", v = "map", w = f.width().unwrap_or_default())
246                    }
247                    _ => write!(
248                        f,
249                        "map<{v}{c:<w$}",
250                        v = blanket,
251                        c = ">",
252                        w = f
253                            .width()
254                            .unwrap_or_default()
255                            .saturating_sub(blanket.min_width() + 4)
256                    ),
257                }
258            }
259        }
260    }
261}

263impl FromStr for MartianBlanketType {
264    type Err = Error;
265    fn from_str(s: &str) -> Result<Self, Self::Err> {
266        if s.ends_with("[]") {
267            // array
268            let t = s.get(0..s.len() - 2).unwrap();
269            Ok(MartianBlanketType::Array(Box::new(
270                MartianBlanketType::from_str(t)?,
271            )))
272        } else if s.starts_with("map<") && s.ends_with('>') {
273            // typed map
274            let t = s.get(4..s.len() - 1).unwrap();
275            Ok(MartianBlanketType::TypedMap(Box::new(
276                MartianBlanketType::from_str(t)?,
277            )))
278        } else {
279            Ok(MartianBlanketType::Primary(MartianPrimaryType::from_str(
280                s,
281            )?))
282        }
283    }
284}

387/// Each variable that is listed in the mro along with it's type form
388/// a `MroField`.
389///
390/// For example, the following stage:
391/// ```mro
392/// stage SORT_ITEMS(
393///     in  int[] unsorted,
394///     in  bool  reverse,
395///     out int[] sorted,
396///     src comp  "my_stage martian sort_items",
397/// )
398/// ```
399/// contains 3 `MroFields`
400/// - MroField { name: "unsorted", ty: MartianBlanketType::Array(MartianPrimaryType::Int)}
401/// - MroField { name: "reverse", ty: MartianBlanketType::Primary(MartianPrimaryType::Bool)}
402/// - MroField { name: "sorted", ty: MartianBlanketType::Array(MartianPrimaryType::Int)}
403#[derive(Debug, Serialize, Clone, Deserialize, PartialEq, Eq, Hash)]
404pub struct MroField {
405    name: String,
406    ty: MartianBlanketType,
407    desc: Option<String>,
408    mro_filename: Option<String>,
409    retain: bool,
410}
411
412impl Display for MroField {
413    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
414        write!(
415            f,
416            "{ty:<width$} {name}",
417            ty = self.ty,
418            width = f.width().unwrap_or_default(),
419            name = self.name.as_str()
420        )
421    }
422}

431impl MroField {
432    /// Create a new `MroField` with the given name and type.
433    /// The field has a default `retain = false`.
434    pub fn new(
435        name: impl ToString,
436        ty: MartianBlanketType,
437        desc: Option<String>,
438        mro_filename: Option<String>,
439    ) -> Self {
440        fn _new_field(
441            name: String,
442            ty: MartianBlanketType,
443            desc: Option<String>,
444            mro_filename: Option<String>,
445        ) -> MroField {
446            let field = MroField {
447                name,
448                ty,
449                desc,
450                mro_filename,
451                retain: false,
452            };
453            field.verify(); // No use case to resultify this so far
454            field
455        }
456        _new_field(name.to_string(), ty, desc, mro_filename)
457    }
458
459    fn name_width(&self) -> usize {
460        self.name.len()
461    }
462
463    fn desc_width(&self) -> usize {
464        match self.desc {
465            Some(ref desc) => {
466                if self.mro_filename.is_some() {
467                    desc.len() + 2 // string length + 2 quote chars
468                } else {
469                    desc.len() + 3 // string length + 2 quote chars + 1 comma
470                }
471            }
472            None => 0,
473        }
474    }
475
476    /// Special formatter that can correctly align all 4 fields that may be present
477    /// note that ty_width, name_width and desc_width are the max width of those columns
478    /// across numerous other MroFields -- e.g. when printing multiple fields of a struct.
479    fn fmt_align_4_columns(&self, ty_width: usize, name_width: usize, desc_width: usize) -> String {
480        let desc_field = match self.desc {
481            Some(ref desc) => format!(" \"{desc}\""),
482            None => "".to_string(),
483        };
484        let mro_filename_field = match self.mro_filename {
485            Some(ref name) => format!(" \"{name}\""),
486            None => "".to_string(),
487        };
488
489        let adjusted_name_width = if self.mro_filename.is_none() && self.desc.is_none() {
490            0
491        } else {
492            name_width
493        };
494
495        let adjusted_desc_width = if self.mro_filename.is_none() {
496            0
497        } else {
498            desc_width
499        };
500
501        format!(
502            "{ty:<ty_width$} {name:<adjusted_name_width$}{desc_field:<adjusted_desc_width$}{mro_filename_field}",
503            ty = self.ty,
504            ty_width = ty_width,
505            adjusted_name_width = adjusted_name_width,
506            adjusted_desc_width = adjusted_desc_width,
507            name = self.name.as_str()
508        )
509    }
510
511    /// Create a new `MroField` with the given name and type, with the
512    /// `retain` field set to true
513    pub fn retained(
514        name: impl ToString,
515        ty: MartianBlanketType,
516        desc: Option<String>,
517        mro_filename: Option<String>,
518    ) -> Self {
519        let mut field = Self::new(name, ty, desc, mro_filename);
520        field.retain = true;
521        field
522    }
523    // Check that name does not match any martian token.
524    fn verify(&self) {
525        for &token in MARTIAN_TOKENS {
526            assert!(
527                self.name != token,
528                "Martian token {} cannot be used as field name",
529                token
530            );
531        }
532        assert!(!self.name.starts_with("__"));
533    }
534}

593const TAB_WIDTH_FOR_MRO: usize = 4;
594macro_rules! mro_using {
595    ($($property:ident: $type:ty),*) => {
596        /// Stuff that comes in the `using` section of a stage definition
597        ///
598        /// For example:
599        /// ```mro
600        /// using (
601        ///     mem_gb  = 4,
602        ///     threads = 16,
603        /// )
604        /// ```
605        #[derive(Debug, Default, Clone, Serialize, Deserialize)]
606        pub struct MroUsing {
607            $(pub $property: Option<$type>,)*
608        }
609
610        impl MroUsing {
611            /// If all fields as None, return False
612            pub fn need_using(&self) -> bool {
613                !($(self.$property.is_none())&&*)
614            }
615        }
616
617        impl Display for MroUsing {
618            fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
619                let field_width = f.width().unwrap_or_else(|| self.min_width());
620                // If every field is None, return empty String
621                if !self.need_using() {
622                    return Ok(());
623                }
624                $(
625                    if let Some($property) = self.$property.clone() {
626                        writeln!(
627                            f,
628                            "{blank:indent$}{key:<width$} = {value},",
629                            blank="",
630                            indent = TAB_WIDTH_FOR_MRO,
631                            key=stringify!($property),
632                            width=field_width,
633                            value=$property
634                        )?;
635                    }
636                )*
637                Ok(())
638            }
639        }
640
641        /// Using section
642        /// ```md
643        /// mem_gb = 1,
644        /// ```
645        impl MroDisplay for MroUsing {
646            fn min_width(&self) -> usize {
647                let mut w = 0;
648                $(if self.$property.is_some() {
649                    w = std::cmp::max(w, stringify!($property).len());
650                })*
651                w
652            }
653        }
654    };
655}

659/// Input and outputs fields together
660#[derive(Debug, Default)]
661pub struct InAndOut {
662    pub inputs: Vec<MroField>,
663    pub outputs: Vec<MroField>,
664}

666impl InAndOut {
667    fn iter_mro_fields(&self) -> impl Iterator<Item = &MroField> {
668        self.inputs.iter().chain(self.outputs.iter())
669    }
670    fn retain_field_names(&self) -> impl Iterator<Item = &str> {
671        self.outputs.iter().filter_map(|field| {
672            if field.retain {
673                Some(field.name.as_ref())
674            } else {
675                None
676            }
677        })
678    }
679}

681impl Display for InAndOut {
682    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
683        let field_width = f.width().unwrap_or_default().max(self.min_width());
684        for (key, fields) in [("in ", &self.inputs), ("out", &self.outputs)] {
685            for field in fields {
686                writeln!(
687                    f,
688                    "{key:>indent$} {value:<width$},",
689                    indent = TAB_WIDTH_FOR_MRO + 3,
690                    key = key,
691                    value = field,
692                    width = field_width
693                )?;
694            }
695        }
696        Ok(())
697    }
698}

740impl FiletypeHeader {
741    /// Find out all the filetypes in the stage and add the extensions
742    /// to the internal hashset which stores all the extensions
743    pub fn add_stage(&mut self, stage_mro: &StageMro) {
744        for field in stage_mro.iter_mro_fields() {
745            self.add_mro_field(field);
746        }
747    }
748    pub fn add_mro_field(&mut self, mro_field: &MroField) {
749        match mro_field.ty.inner() {
750            MartianPrimaryType::FileType(ref ext) => {
751                self.0.insert(ext.to_string());
752            }
753            MartianPrimaryType::Struct(ref def) => {
754                for field in &def.fields {
755                    self.add_mro_field(field);
756                }
757            }
758            _ => {}
759        }
760    }
761}

769        extensions.sort();
770        for ext in extensions {
771            writeln!(f, "filetype {ext};")?;
772        }
773        writeln!(f)
774    }

785/// All the structs that need to be defined in an mro
786#[derive(Debug, Default)]
787pub struct StructHeader(BTreeMap<String, (StructDef, usize)>); // key = struct name, val = (struct def, insertion index)
788
789impl From<&StageMro> for StructHeader {
790    fn from(stage_mro: &StageMro) -> StructHeader {
791        let mut result = StructHeader(BTreeMap::new());
792        result.add_stage(stage_mro);
793        result
794    }
795}
796
797impl StructHeader {
798    /// Find out all the structs in the stage and add it to the
799    /// internal hashmap which stores all the structs
800    pub fn add_stage(&mut self, stage_mro: &StageMro) {
801        for field in stage_mro.iter_mro_fields() {
802            self.add_mro_field(field);
803        }
804    }
805    pub fn add_mro_field(&mut self, mro_field: &MroField) {
806        if let MartianPrimaryType::Struct(ref def) = mro_field.ty.inner() {
807            for field in &def.fields {
808                self.add_mro_field(field);
809            }
810            if self.0.contains_key(&def.name) {
811                assert_eq!(
812                    &self.0[&def.name].0, def,
813                    "struct {} has conflicting definitions.\nDefinition 1: {:?}\nDefinition 2: {def:?}",
814                    def.name, &self.0[&def.name]
815                );
816            } else {
817                let index = self.0.len();
818                self.0.insert(def.name.clone(), (def.clone(), index));
819            }
820        }
821    }
822}
823
824impl MroDisplay for StructHeader {
825    fn min_width(&self) -> usize {
826        0
827    }
828}

830impl Display for StructHeader {
831    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
832        let field_width = f.width().unwrap_or_else(|| self.min_width());
833
834        let mut struct_defs: Vec<(StructDef, usize)> = self.0.values().cloned().collect();
835        struct_defs.sort_by_key(|x| x.1);
836
837        for struct_def in struct_defs.iter() {
838            writeln!(
839                f,
840                "{def:<field_width$}",
841                def = struct_def.0,
842                field_width = field_width
843            )?;
844        }
845        Ok(())
846    }

849/// An object that can generate a `StageMro`
850///
851/// Can be auto generated using proc macro attribute
852/// `#[make_mro]` on MartianMain or MartianStage
853/// implementations if the associated types implement `MartianStruct`
854pub trait MroMaker {
855    fn stage_mro(adapter_name: impl ToString, stage_key: impl ToString) -> StageMro {
856        let result = StageMro {
857            stage_name: Self::stage_name(),
858            adapter_name: adapter_name.to_string(),
859            stage_key: stage_key.to_string(),
860            stage_in_out: Self::stage_in_and_out(),
861            chunk_in_out: Self::chunk_in_and_out(),
862            using_attrs: Self::using_attributes(),
863        };
864        result.verify();
865        result
866    }
867    fn mro(adapter_name: impl ToString, stage_key: impl ToString) -> String {
868        let stage_mro = Self::stage_mro(adapter_name, stage_key);
869        let filetype = FiletypeHeader::from(&stage_mro);
870        let struct_header = StructHeader::from(&stage_mro);
871        format!("{filetype}{struct_header}{stage_mro}")
872    }
873    fn stage_name() -> &'static str;
874    fn stage_in_and_out() -> InAndOut;
875    fn chunk_in_and_out() -> Option<InAndOut>;
876    fn using_attributes() -> MroUsing;
877}
878
879/// All the data needed to create a stage definition mro.
880#[derive(Debug)]
881pub struct StageMro {
882    stage_name: &'static str, // e.g CORRECT_BARCODES in `stage CORRECT_BARCODES(..)`
883    adapter_name: String, // Martian adapter e.g `cr_slfe` in `src comp "cr_slfe martian correct_barcodes"
884    stage_key: String, // Key used in the hashmap containing all stages e.g `correct_barcodes` in `src comp "cr_slfe martian correct_barcodes"
885    stage_in_out: InAndOut, // Inputs and outputs of the stage
886    chunk_in_out: Option<InAndOut>, // Inputs and outputs of the chunk. None indicates a stage with only a main
887    using_attrs: MroUsing,          // Things coming under using
888}
889
890impl MroDisplay for StageMro {
891    fn min_width(&self) -> usize {
892        0
893    }
894}
895
896impl Display for StageMro {
897    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
898        // Determing the field width for the type field
899        let ty_width = std::cmp::max(
900            self.stage_in_out.min_width(),
901            self.chunk_in_out
902                .as_ref()
903                .map(MroDisplay::min_width)
904                .unwrap_or_default(),
905        );
906        writeln!(f, "stage {}(", self.stage_name)?;
907        write!(
908            f,
909            "{params:<ty_width$}",
910            params = self.stage_in_out,
911            ty_width = ty_width
912        )?;
913        writeln!(
914            f,
915            r#"{blank:indent$}src {comp:<ty_width$} "{adapter} martian {stage_key}","#,
916            blank = "",
917            indent = TAB_WIDTH_FOR_MRO,
918            comp = "comp",
919            ty_width = ty_width,
920            adapter = self.adapter_name,
921            stage_key = self.stage_key,
922        )?;
923
924        if let Some(ref chunk_in_out) = self.minified_chunk_in_outs() {
925            writeln!(f, ") split (")?;
926            write!(
927                f,
928                "{params:<ty_width$}",
929                params = chunk_in_out,
930                ty_width = ty_width
931            )?;
932        }
933
934        if self.using_attrs.need_using() {
935            writeln!(f, ") using (")?;
936            write!(f, "{}", self.using_attrs)?;
937        }
938        let mut retain_names = self.stage_in_out.retain_field_names();
939        if let Some(first) = retain_names.next() {
940            writeln!(f, ") retain (")?;
941            writeln!(
942                f,
943                "{blank:indent$}{line},",
944                blank = "",
945                indent = TAB_WIDTH_FOR_MRO,
946                line = first
947            )?;
948            for line in retain_names {
949                writeln!(
950                    f,
951                    "{blank:indent$}{line},",
952                    blank = "",
953                    indent = TAB_WIDTH_FOR_MRO,
954                    line = line
955                )?;
956            }
957        }
958        writeln!(f, ")")
959    }
960}
961
962impl StageMro {
963    fn iter_mro_fields(&self) -> impl Iterator<Item = &MroField> {
964        self.stage_in_out
965            .iter_mro_fields()
966            .chain(self.chunk_in_out.iter().flat_map(InAndOut::iter_mro_fields))
967    }
968    fn minified_chunk_in_outs(&self) -> Option<InAndOut> {
969        match self.chunk_in_out {
970            Some(ref chunk_in_out) => {
971                let chunk_in = chunk_in_out.inputs.clone();
972                let skip_names: HashSet<_> = self
973                    .stage_in_out
974                    .outputs
975                    .iter()
976                    .map(|field| &field.name)
977                    .collect();
978                let chunk_out: Vec<_> = chunk_in_out
979                    .outputs
980                    .iter()
981                    .filter(|field| !skip_names.contains(&field.name))
982                    .cloned()
983                    .collect();
984                Some(InAndOut {
985                    inputs: chunk_in,
986                    outputs: chunk_out,
987                })
988            }
989            None => None,
990        }
991    }
992    fn verify(&self) {
993        // By design, all the field names are guaranteed to be not
994        // any of the martian tokens. It raises a compile error when
995        // deriving MartianStruct and is checked when creating a
996        // MaroField wusing new() which is the only public entry point.
997        // So we don't have anything to check for a MainOnly stage
998        let chunk_in_out = if let Some(ref chunk_in_out) = self.chunk_in_out {
999            chunk_in_out
1000        } else {
1001            return;
1002        };
1003        // Do not allow the same field name in stage and chunk inputs
1004        // O(mn) is good enough
1005        for f_chunk in chunk_in_out.inputs.iter() {
1006            for f_stage in self.stage_in_out.inputs.iter() {
1007                assert!(
1008                    f_chunk.name != f_stage.name,
1009                    "ERROR: Found identical field {} in stage and chunk inputs, which is not allowed",
1010                    f_chunk.name
1011                )
1012            }
1013        }
1014
1015        // Fields having same name in stage and chunk outputs
1016        // should have identical types.
1017        // O(mn) is good enough
1018        for f_chunk in chunk_in_out.outputs.iter() {
1019            for f_stage in self.stage_in_out.outputs.iter() {
1020                if f_chunk.name == f_stage.name {
1021                    assert!(
1022                        f_chunk.ty == f_stage.ty,
1023                        "ERROR: Found identical field {} in stage and chunk outputs, but it has type {} in stage outputs and type {} in chunk outputs.",
1024                        f_chunk.name, f_stage.ty, f_chunk.ty,
1025                    )
1026                }
1027            }
1028        }
1029    }
1030}
