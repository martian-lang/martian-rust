51#[cold]
52fn write_errors(msg: &str, is_assert: bool) -> Result<()> {
53    let mut err_file: File = unsafe { File::from_raw_fd(4) };
54
55    // We want to aggressively avoid allocations here if we can, since one
56    // common source of errors is running out of memory.
57    let msg_alloc: String;
58    let msg = if is_assert {
59        msg_alloc = ["ASSERT:", msg].concat();
60        msg_alloc.as_str()
61    } else {
62        msg
63    };
64
65    let _ = err_file.write_all(msg.as_bytes());
66
67    // Avoid closing err_file
68    let _ = err_file.into_raw_fd();
69    Ok(())
70}

72// e.g. 2006-01-02 15:04:05.  Note that this is only crate-public, not fully
73// public, because this is a bad date format that is used only to retain
74// backwards compatibility.  For use cases where that is not a concern, use
75// Rfc3339, which looks similar but won't cause strange bugs if not every
76// compute node in your cluster is using the same time zone.
77//
78// We could use the proc macro, but then we'd need
79// to compile the proc macro crate, which would slow down build times
80// significantly for very little benefit in readability.
81pub(crate) const DATE_FORMAT: &[FormatItem<'_>] = &[
82    FormatItem::Component(Component::Year(Year::default())),
83    Literal(b"-"),
84    FormatItem::Component(Component::Month(Month::default())),
85    Literal(b"-"),
86    FormatItem::Component(Component::Day(Day::default())),
87    Literal(b" "),
88    FormatItem::Component(Component::Hour(Hour::default())),
89    Literal(b":"),
90    FormatItem::Component(Component::Minute(Minute::default())),
91    Literal(b":"),
92    FormatItem::Component(Component::Second(Second::default())),
93];

95fn setup_logging(log_file: File, level: LevelFilter) {
96    let base_config = fern::Dispatch::new().level(level);
97
98    let logger_config = fern::Dispatch::new()
99        .format(|out, msg, record| {
100            let time_str = OffsetDateTime::now_local()
101                .unwrap_or_else(|_| OffsetDateTime::now_utc())
102                .format(DATE_FORMAT)
103                .unwrap();
104            out.finish(format_args!("{time_str} [{}] {msg}", record.target()))
105        })
106        .chain(log_file)
107        .chain(io::stdout());
108
109    let cfg = base_config.chain(logger_config).apply();
110
111    if let Err(e) = cfg {
112        panic!("Failed to initialize global logger: {e}");
113    }
114}

116/// Configure the Martian adapter for executing stage code
117pub struct MartianAdapter<S> {
118    stage_map: HashMap<String, Box<dyn RawMartianStage>, S>,
119    log_level: LevelFilter,
120    is_error_assert: Box<dyn (Fn(&Error) -> bool) + 'static>,
121}
122
123impl<S: std::hash::BuildHasher> MartianAdapter<S> {
124    /// Build a new Martian adapter with the given registry of Martian stages
125    /// Arguments:
126    ///  - `stage_map`: names and implementations of the Martian stages that can be run by this binary.
127    pub fn new(stage_map: HashMap<String, Box<dyn RawMartianStage>, S>) -> MartianAdapter<S> {
128        MartianAdapter {
129            stage_map,
130            log_level: LevelFilter::Warn,
131            is_error_assert: Box::new(|_| false),
132        }
133    }
134
135    /// Set the minimum severity level of log messages that are emitted to the Martian
136    /// _log file.
137    pub fn log_level(self, log_level: LevelFilter) -> MartianAdapter<S> {
138        MartianAdapter { log_level, ..self }
139    }
140
141    ///  Set `is_error_assert`, predicate determining whether to emit an error as an ASSERT
142    ///  to Martian. ASSERT errors indicate an unrecoverable configuration error, and will
143    ///  prevent the user from restarting the pipeline. The is_error_assert function should
144    ///  use downcasting to match the error against a set of error types that should generate an assert.
145    pub fn assert_if<F: 'static + Fn(&Error) -> bool>(self, predicate: F) -> MartianAdapter<S> {
146        MartianAdapter {
147            is_error_assert: Box::new(predicate),
148            ..self
149        }
150    }
151
152    /// Run the martian adapter using the given cmdline args
153    /// provided by the martian runtime. The caller should call sys::exit() witih
154    /// the returncode returned by this function.
155    /// Arguments:
156    ///  - `args`: vector of command line arguments, typically supplied by Martian runtime.
157    #[must_use = "Martian stage binaries should call std::process::exit() on the return_code"]
158    pub fn run(self, args: Vec<String>) -> i32 {
159        self.run_get_error(args).0
160    }
161
162    /// Like `run()` but also return an error thrown by the stage (if any). May be useful
163    /// for unit testing purposes.
164    #[must_use = "Martian stage binaries should call std::process::exit() on the return_code"]
165    pub fn run_get_error(self, args: Vec<String>) -> (i32, Option<Error>) {
166        martian_entry_point(args, self.stage_map, self.log_level, self.is_error_assert)
167    }
168}

170/// See docs on MartianAdapter methods for details.
171fn martian_entry_point<S: std::hash::BuildHasher>(
172    args: Vec<String>,
173    stage_map: HashMap<String, Box<dyn RawMartianStage>, S>,
174    level: LevelFilter,
175    is_error_assert: Box<dyn Fn(&Error) -> bool>,
176) -> (i32, Option<Error>) {
177    info!("got args: {:?}", args);
178
179    // turn on backtrace capture
180    std::env::set_var("RUST_BACKTRACE", "1");
181
182    // Hook rust logging up to Martian _log file
183    let log_file: File = unsafe { File::from_raw_fd(3) };
184    setup_logging(log_file, level);
185
186    // setup Martian metadata
187    // special handler for error in stage setup
188    let md = match initialize(args).context("IO Error initializing stage") {
189        Ok(m) => m,
190        Err(e) => {
191            let _ = write_errors(&format!("{e:?}"), false);
192            return (1, Some(e));
193        }
194    };
195
196    // Get the stage implementation
197    let stage = stage_map.get(&md.stage_name).with_context(
198        #[cold]
199        || format!("Couldn't find requested Martian stage: {}", md.stage_name),
200    );
201
202    // special handler for non-existent stage
203    let stage = match stage {
204        Ok(s) => s,
205        Err(e) => {
206            let _ = write_errors(&format!("{e:?}"), false);
207            return (1, Some(e));
208        }
209    };
210
211    // will write to this from panic handler if needed.
212    // panic handler has separate write code to avoid locking.
213    let stackvars_path = md.make_path("stackvars");
214
215    // Setup panic hook. If a stage panics, we'll shutdown cleanly to martian
216    let p = panic::take_hook();
217    panic::set_hook(Box::new(
218        #[cold]
219        move |info| {
220            let backtrace = Backtrace::new();
221
222            let msg = match info.payload().downcast_ref::<&'static str>() {
223                Some(&s) => s,
224                None => match info.payload().downcast_ref::<String>() {
225                    Some(s) => (*s).as_str(),
226                    None => "Box<Any>",
227                },
228            };
229
230            let msg = match info.location() {
231                Some(location) => format!(
232                    "stage failed unexpectedly: '{}' {}:{}:\n{:?}",
233                    msg,
234                    location.file(),
235                    location.line(),
236                    backtrace
237                ),
238                None => format!("stage failed unexpectedly: '{msg}':\n{backtrace:?}"),
239            };
240
241            // write to _log
242            error!("{}", msg);
243
244            // write stack trace to to _stackvars.
245            // this will just give up if any errors are encountere
246            let bt_string = format!("{backtrace:?}");
247            let _ = File::create(&stackvars_path).map(move |mut f| {
248                let _ = f.write_all(bt_string.as_bytes());
249            });
250
251            // write to _errors
252            let _ = write_errors(&msg, false);
253
254            // call default panic handler (not sure if this is a good idea or not)
255            p(info);
256        },
257    ));
258
259    let stage_type = md.stage_type;
260
261    let md = Arc::new(Mutex::new(md));
262
263    let result = match stage_type {
264        StageType::Split => stage.split(md.clone()),
265        StageType::Main => stage.main(md.clone()),
266        StageType::Join => stage.join(md.clone()),
267    };
268
269    match result {
270        // exit code = 0
271        Ok(()) => (0, None),
272
273        // write message and stack trace, exit code = 1;
274        Err(e) => {
275            report_error(&mut md.lock().unwrap(), &e, is_error_assert(&e));
276            (1, Some(e))
277        }
278    }
279}

281#[cold]
282fn report_error(md: &mut Metadata, e: &Error, is_assert: bool) {
283    let bt = e.backtrace();
284    let _ = md.stackvars(&bt.to_string());
285    let _ = write_errors(&format!("{e:#}"), is_assert);
286}

293/// Write MRO to filename or stdout.
294pub fn martian_make_mro(
295    header_comment: &str,
296    filename: Option<impl AsRef<Path>>,
297    rewrite: bool,
298    mro_registry: Vec<StageMro>,
299) -> Result<()> {
300    if let Some(filename) = &filename {
301        let filename = filename.as_ref();
302        ensure!(
303            !filename.is_dir(),
304            "Path {} is a directory",
305            filename.display()
306        );
307        ensure!(
308            rewrite || !filename.exists(),
309            "File {} exists. Use --rewrite to overwrite it.",
310            filename.display()
311        );
312    }
313
314    let mro = make_mro_string(header_comment, &mro_registry);
315    match filename {
316        Some(filename) => {
317            let filename = filename.as_ref();
318            File::create(filename)
319                .with_context(|| filename.display().to_string())?
320                .write_all(mro.as_bytes())
321                .with_context(|| filename.display().to_string())?;
322        }
323        None => {
324            print!("{mro}");
325        }
326    }
327    Ok(())
328}

330pub fn make_mro_string(header_comment: &str, mro_registry: &[StageMro]) -> String {
331    let mut filetype_header = FiletypeHeader::default();
332    let mut struct_header = StructHeader::default();
333    let mut mro_string = String::new();
334    for stage_mro in mro_registry {
335        filetype_header.add_stage(stage_mro);
336        struct_header.add_stage(stage_mro);
337        writeln!(&mut mro_string, "{stage_mro}").unwrap();
338    }
339    mro_string.pop();
340
341    if header_comment.is_empty() {
342        format!(
343            "#
344# Code generated by {}.  DO NOT EDIT.
345#
346
347{}{}{}",
348            get_generator_name(),
349            filetype_header,
350            struct_header,
351            mro_string
352        )
353    } else {
354        assert!(
355            header_comment
356                .lines()
357                .all(|line| line.trim_end().is_empty() || line.starts_with('#')),
358            "All non-empty header lines must start with '#', but got\n{header_comment}"
359        );
360        format!(
361            "{}
362#
363# Code generated by {}.  DO NOT EDIT.
364#
365
366{}{}{}",
367            header_comment,
368            get_generator_name(),
369            filetype_header,
370            struct_header,
371            mro_string
372        )
373    }
