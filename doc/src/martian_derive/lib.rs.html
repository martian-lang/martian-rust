26/// When this attribute is applied to the `MartianMain` or `MartianStage` trait implementation of
27/// a stage struct, it derives the trait `MroMaker` to the stage struct, which lets you generate
28/// the mro corresponding to the stage.
29///
30/// You can optionally specify `mem_gb`, `threads`, `vmem_gb` and `volatile` within this proc-macro.
31/// For example, use `#[make_mro(mem_gb = 4, threads = 2]` for setting `mem_gb` and `threads` that would
32/// appear in the `using()` section of the mro definition.
33///
34/// You can also set the stage name here. By default, the stage name in the mro is the SHOUTY_SNAKE_CASE version
35/// of the stage struct name. You can override that using: `#[make_mro(mem_gb = 2, stage_name = MY_CUSTOM_NAME)]`
36///
37/// For examples on how to use it and customize, take a look at `tests/test_full_mro.rs`
38#[proc_macro_attribute]
39pub fn make_mro(
40    attr: proc_macro::TokenStream,
41    item: proc_macro::TokenStream,
42) -> proc_macro::TokenStream {
43    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
44    // STEP 0
45    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
46    // Parse the attributes and emit error if it is invalid
47    // Attributes are stuff inside the bracket after make_mro in
48    // `#[make_mro(mem_gb = 4, threads = 2)]`
49    // If we get valid attributes, create `fn using_attributes() -> MroUsing;`
50    let parsed_attr: MakeMroAttr = match attr.to_string().parse() {
51        Ok(parsed) => parsed,
52        Err(e) => {
53            let attr2 = proc_macro2::TokenStream::from(attr);
54            return syn::Error::new_spanned(attr2, e).to_compile_error().into();
55        }
56    };
57
58    // If a stage name is specified, make sure it's in SHOUTY_SNAKE_CASE
59    if let Some(ref name) = parsed_attr.stage_name {
60        let expected = utils::to_shouty_snake_case(name);
61        if expected != *name {
62            let span = attr
63                .into_iter()
64                .find(|tt| {
65                    if let proc_macro::TokenTree::Ident(ref ident) = tt {
66                        ident.to_string() == "stage_name"
67                    } else {
68                        false
69                    }
70                })
71                .unwrap()
72                .span()
73                .into();
74            return syn::Error::new(
75                span,
76                format!(
77                    "`stage_name` needs to be in SHOUTY_SNAKE_CASE (without any surrounding quotes). \
78                    Found {name}, use {expected} instead",
79                ),
80            )
81            .to_compile_error()
82            .into();
83        }
84    }
85
86    let mem_gb_quote = parsed_attr
87        .mem_gb
88        .map(|x| quote![mem_gb: Some(#x),])
89        .unwrap_or_default();
90    let vmem_gb_quote = parsed_attr
91        .vmem_gb
92        .map(|x| quote![vmem_gb: Some(#x),])
93        .unwrap_or_default();
94    let threads_quote = parsed_attr
95        .threads
96        .map(|x| quote![threads: Some(#x),])
97        .unwrap_or_default();
98    let volatile_quote = match parsed_attr.volatile {
99        Some(k) => match k {
100            Volatile::Strict => quote![volatile: Some(::martian::Volatile::Strict),],
101            Volatile::False => quote![volatile: Some(::martian::Volatile::False),],
102        },
103        None => quote![volatile: None,],
104    };
105    let using_attributes_fn = quote![
106        fn using_attributes() -> ::martian::MroUsing {
107            #[allow(clippy::needless_update)]
108            ::martian::MroUsing {
109                #mem_gb_quote
110                #threads_quote
111                #vmem_gb_quote
112                #volatile_quote
113                ..Default::default()
114            }
115        }
116    ];
117
118    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
119    // STEP 1
120    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
121    // Make sure that #[make_mro] attribute is used on a trait impl and not
122    // anything else.
123    // The way we achieve it is to try parsing the input TokenStrean as `ItemImpl`
124    // and checking the parse result
125    let item_impl = match syn::parse::<ItemImpl>(item.clone()) {
126        Ok(item_impl) => item_impl,
127        Err(_) => {
128            let span = proc_macro2::TokenStream::from(item);
129            return syn::Error::new_spanned(span, ATTR_NOT_ON_TRAIT_IMPL_ERROR)
130                .to_compile_error()
131                .into();
132        }
133    };
134
135    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
136    // STEP 2
137    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
138    // Identify whether #[make_mro] was applied to a `MartianMain` or `MartianStage`
139    // If we find that it was applied to a different trait impl, produce a
140    // sensible compile error. This only checks for the trait by name, so it is possible
141    // to trick the compiler to continue, but it should fail later if the trait
142    // signature is different from the one defined in `martian` crate
143    let trait_path = item_impl.trait_.unwrap().1;
144    let which_trait = match parse_which_trait(trait_path) {
145        Ok(t) => t,
146        Err(e) => return e.to_compile_error().into(),
147    };
148
149    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
150    // STEP 3
151    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
152    // Now we know which trait the attribute has been applied to. Find out the
153    // name of the stage struct. We need to implement `MakeMro` trait for this
154    // struct. We can build the `stage_name()` function from the struct name.
155    // TODO: This might also be a good time to check that the implementation is
156    // actually for a unit struct. But is it possible?
157    let stage_struct = item_impl.self_ty;
158    let stage_struct_name = match *stage_struct {
159        Type::Path(ref ty_path) => {
160            let segments = &ty_path.path.segments;
161            segments.iter().last().unwrap().ident.to_string()
162        }
163        _ => {
164            let span = proc_macro2::TokenStream::from(item);
165            return syn::Error::new_spanned(span, "Expecting the impl for a struct.")
166                .to_compile_error()
167                .into();
168        }
169    };
170    // Stage name is the name of the stage in the mro. It can either be set using
171    // the name attribute or can be derived by converting the struct to shouting
172    // snake case
173    let stage_name =
174        utils::to_shouty_snake_case(&parsed_attr.stage_name.unwrap_or(stage_struct_name));
175    let stage_name_fn = quote![
176        fn stage_name() -> &'static str {
177            #stage_name
178        }
179    ];
180
181    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
182    // STEP 4
183    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
184    // Find out the associated types `StageInputs`, `StageOutputs`,
185    // `ChunkInputs`, `ChunkOutputs` and meke the two finctions
186    // - `fn stage_in_and_out() -> InAndOut;`
187    // - `fn chunk_in_and_out() -> Option<InAndOut>;`
188    let mut builder = AssociatedTypeBuilder::default();
189    for impl_item in &item_impl.items {
190        if let ImplItem::Type(ty) = impl_item {
191            builder.add(ty.ident.clone(), ty.ty.clone());
192        }
193    }
194    let stage_var_fn = builder.to_quote(which_trait);
195
196    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
197    // STEP 5
198    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
199    // Stitch the quotes together
200    let (impl_generics, _, where_clause) = item_impl.generics.split_for_impl();
201    let item_clone2 = proc_macro2::TokenStream::from(item);
202    quote![
203        #item_clone2
204        #[automatically_derived]
205        impl #impl_generics ::martian::MroMaker for #stage_struct #where_clause {
206            #stage_var_fn
207            #stage_name_fn
208            #using_attributes_fn
209        }
210    ]
211    .into()
212}

214#[derive(Default)]
215struct AssociatedTypeBuilder {
216    stage_inputs: Option<Type>,
217    stage_outputs: Option<Type>,
218    chunk_inputs: Option<Type>,
219    chunk_outputs: Option<Type>,
220}
221
222impl AssociatedTypeBuilder {
223    fn add(&mut self, associated_type_key: Ident, associated_type_val: Type) {
224        match associated_type_key.to_string().as_ref() {
225            STAGE_INPUT_IDENT => {
226                if self.stage_inputs.is_some() {
227                    unreachable!("Found multiple assignments for associated type {}. I am not sure how to proceed.", STAGE_INPUT_IDENT)
228                }
229                self.stage_inputs = Some(associated_type_val);
230            }
231            STAGE_OUTPUT_IDENT => {
232                if self.stage_outputs.is_some() {
233                    unreachable!("Found multiple assignments for associated type {}. I am not sure how to proceed.", STAGE_OUTPUT_IDENT)
234                }
235                self.stage_outputs = Some(associated_type_val);
236            }
237            CHUNK_INPUT_IDENT => {
238                if self.chunk_inputs.is_some() {
239                    unreachable!(
240                        "Found multiple assignments for associated type {}. \
241                    I am not sure how to proceed.",
242                        CHUNK_INPUT_IDENT
243                    )
244                }
245                self.chunk_inputs = Some(associated_type_val);
246            }
247            CHUNK_OUTPUT_IDENT => {
248                if self.chunk_outputs.is_some() {
249                    unreachable!(
250                        "Found multiple assignments for associated type {}. \
251                    I am not sure how to proceed.",
252                        CHUNK_INPUT_IDENT
253                    )
254                }
255                self.chunk_outputs = Some(associated_type_val);
256            }
257            s => unreachable!(
258                "Got an unexpected associated type {}. \
259                I am not sure how to proceed",
260                s
261            ),
262        }
263    }
264    fn to_quote(&self, which: StageKind) -> proc_macro2::TokenStream {
265        assert!(self.stage_inputs.is_some());
266        assert!(self.stage_outputs.is_some());
267        let si = self.stage_inputs.clone().unwrap();
268        let so = self.stage_outputs.clone().unwrap();
269        let stage_in_out_fn = quote![
270            fn stage_in_and_out() -> ::martian::InAndOut {
271                ::martian::InAndOut {
272                    inputs: <#si as ::martian::MartianStruct>::mro_fields(),
273                    outputs: <#so as ::martian::MartianStruct>::mro_fields(),
274                }
275            }
276        ];
277        let chunk_in_out_fn = match which {
278            StageKind::MainOnly => {
279                assert!(self.chunk_inputs.is_none());
280                assert!(self.chunk_outputs.is_none());
281                quote![
282                    fn chunk_in_and_out() -> Option<::martian::InAndOut> {
283                        None
284                    }
285                ]
286            }
287            StageKind::WithSplit => {
288                assert!(self.chunk_inputs.is_some());
289                assert!(self.chunk_outputs.is_some());
290                let ci = self.chunk_inputs.clone().unwrap();
291                let co = self.chunk_outputs.clone().unwrap();
292                quote![
293                    fn chunk_in_and_out() -> Option<::martian::InAndOut> {
294                        Some(::martian::InAndOut {
295                            inputs: <#ci as ::martian::MartianStruct>::mro_fields(),
296                            outputs: <#co as ::martian::MartianStruct>::mro_fields(),
297                        })
298                    }
299                ]
300            }
301        };
302        quote![
303            #stage_in_out_fn
304            #chunk_in_out_fn
305        ]
306    }
307}

309// Identify which trait impl the attribute is applied to among `MartianMain`
310// and `MartianStage`. If we find that this is applied to a different trait,
311// return an error.
312fn parse_which_trait(trait_path: syn::Path) -> Result<StageKind, Error> {
313    let mut last_ident = String::from("");
314    let span = trait_path.segments[0].ident.span();
315    for segment in trait_path.segments {
316        if segment.ident == MARTIAN_MAIN_TRAIT {
317            return Ok(StageKind::MainOnly);
318        }
319        if segment.ident == MARTIAN_STAGE_TRAIT {
320            return Ok(StageKind::WithSplit);
321        }
322        last_ident = segment.ident.to_string();
323    }
324
325    Err(Error::new(
326        span,
327        format!(
328            "{}. You are trying to use it on {} trait implementation.",
329            ATTR_NOT_ON_TRAIT_IMPL_ERROR, last_ident
330        ),
331    ))
332}

334// A macro to create MakeMroAttr struct which will parse a comma separated
335// `key=value` pairs. The input to the macro is the list of keys and their
336// type. All they keys are optional and the type can be any type which can be
337// parsed
338macro_rules! attr_parse {
339    ($($property:ident: $type:ty),*) => {
340        #[derive(Debug, Default, PartialEq)]
341        struct MakeMroAttr {
342            $($property: Option<$type>,)*
343        }
344        impl FromStr for MakeMroAttr {
345            type Err = String;
346            fn from_str(s: &str) -> Result<Self, Self::Err> {
347                // When we specify negative values, for e.g #[make_mro(threads = -4)]
348                // casting the attribute Tokenstream to String intoduces an additional
349                // space after `-`, i.e we get "threads=- 4". So we get rid of the
350                // whitespaces here
351                let s: String = s.split_whitespace().collect();
352                if s.is_empty() {
353                    return Ok(MakeMroAttr::default());
354                }
355                $(let mut $property = None;)*
356                for using_spec in s.split(',') {
357                    let parts: Vec<_> = using_spec.trim().splitn(3, '=').map(str::trim).collect();
358                    if parts.len() != 2 {
359                        return Err(format!(
360                            "Expecting a comma separated `key=value` like tokens here. \
361                            The allowed keys are: [{}]",
362                            stringify!($($property),*)));
363                    }
364                    match parts[0] {
365                        $(stringify!($property) => {
366                            // Make sure that this is the first assignment
367                            if $property.is_some() {
368                                return Err(format!("Found multiple assignments for {}.", parts[0]))
369                            }
370                            $property = match parts[1].parse::<$type>() {
371                                Ok(parsed) => Some(parsed),
372                                Err(_) => return Err(format!(
373                                    "Unable to parse {0} as {1} from `{0}={2}`",
374                                    parts[0], stringify!($type), parts[1]))
375                            };
376                        },)*
377                        _ => return Err(format!(
378                            "Expecting a comma separated `key=value` like tokens here. \
379                            The allowed keys are: [{}]. Found an invalid key {}",
380                            stringify!($($property),*), parts[0]))
381                    }
382                }
383                Ok(MakeMroAttr {
384                    $($property,)*
385                })
386            }
387        }
388    }
389}

400/// Structs which are used as associated types in `MartianMain` or `MartianStage`
401/// traits need to implement `MartianStruct`. You can derive it using `#[derive(MartianStruct)]`
402/// Each field in the struct needs to implement `AsMartianBlanketType`. This is implemented
403/// for most basic Rust types. You can derive it on custom structs or enum using `#[derive(MartianType)]`.
404///
405/// You can optionally annotate the mro type manually for each field (`#[mro_type = "map[]"]` for example).
406/// This is useful if the datatype for the field is from a third party crate.
407///
408/// You can optionally add a field to the "retain" section of the mro using `#[mro_retain]`.
409///
410#[proc_macro_derive(MartianStruct, attributes(mro_retain, mro_type, mro_filename))]
411pub fn martian_struct(item: proc_macro::TokenStream) -> proc_macro::TokenStream {
412    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
413    // STEP 1
414    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
415    // Make sure that #[derive(MartianStruct)] is used on struct
416    // The way we achieve it is to try parsing the input TokenStrean as `ItemStruct`
417    // and checking the parse result
418    let item_struct = match syn::parse::<ItemStruct>(item.clone()) {
419        Ok(item_struct) => item_struct,
420        Err(_) => {
421            let span = proc_macro2::TokenStream::from(item);
422            return syn::Error::new_spanned(span, MARTIAN_STRUCT_NOT_ON_NAMED_STRUCT_ERROR)
423                .to_compile_error()
424                .into();
425        }
426    };
427
428    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
429    // STEP 2
430    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
431    // Collect the fields of the struct. If we get a unit struct or tuple struct
432    // produce a compile error.
433    let fields = match item_struct.fields.clone() {
434        Fields::Named(f) => f.named,
435        _ => {
436            let span = proc_macro2::TokenStream::from(item);
437            return syn::Error::new_spanned(span, MARTIAN_STRUCT_NOT_ON_NAMED_STRUCT_ERROR)
438                .to_compile_error()
439                .into();
440        }
441    };
442
443    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
444    // STEP 3
445    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
446    // Generate tokenstream for `MroField` calls for each field
447    // Make sure that none of the field names are martian keywords.
448    // Parse the #[mro_retain] attributes attached to the field, and make sure
449    // that no serde field attributes are used
450    let mut vec_inner = Vec::with_capacity(fields.len());
451    let blacklist: HashSet<&str> = MARTIAN_TOKENS.iter().copied().collect();
452    for field in fields {
453        let name = field.ident.clone().unwrap().to_string();
454        let mut retain = false;
455        let mut mro_type = None;
456        let mut doc_comment = None;
457        let mut mro_filename = None;
458
459        for attr in &field.attrs {
460            if attr.path().is_ident("mro_retain") {
461                if !matches!(attr.meta, syn::Meta::Path(_)) {
462                    return syn::Error::new_spanned(field, "mro_retain accepts no arguments")
463                        .to_compile_error()
464                        .into();
465                }
466                retain = true;
467            }
468
469            if attr.path().is_ident("serde") {
470                return syn::Error::new_spanned(
471                    field,
472                    "Cannot use serde attributes here. \
473                     This might be okay, but it's hard to guarantee that deriving \
474                     MartianStruct would work correctly when using serde attributes.",
475                )
476                .to_compile_error()
477                .into();
478            }
479
480            if attr.path().is_ident("mro_type") {
481                let syn::Meta::NameValue(syn::MetaNameValue {
482                    path: _,
483                    eq_token: _,
484                    value:
485                        syn::Expr::Lit(syn::ExprLit {
486                            attrs,
487                            lit: syn::Lit::Str(val),
488                        }),
489                }) = &attr.meta
490                else {
491                    return syn::Error::new_spanned(field, INVALID_MRO_TYPE_ERROR)
492                        .to_compile_error()
493                        .into();
494                };
495
496                if !attrs.is_empty() {
497                    return syn::Error::new_spanned(field, INVALID_MRO_TYPE_ERROR)
498                        .to_compile_error()
499                        .into();
500                }
501
502                if val.value().parse::<MartianBlanketType>().is_err() {
503                    let value = val.value();
504                    return syn::Error::new_spanned(
505                        field,
506                        format!("{INVALID_MRO_TYPE_ERROR}. Found {value}"),
507                    )
508                    .to_compile_error()
509                    .into();
510                }
511
512                if mro_type.is_some() {
513                    return syn::Error::new_spanned(
514                        field,
515                        format!("Specified #[mro_type] twice for field '{name}'"),
516                    )
517                    .to_compile_error()
518                    .into();
519                }
520
521                mro_type = Some(val.value())
522            } else if attr.path().is_ident("doc") && doc_comment.is_none() {
523                if let Meta::NameValue(meta) = &attr.meta {
524                    if let Expr::Lit(elit) = &meta.value {
525                        if let Lit::Str(lstr) = &elit.lit {
526                            let val = lstr.value();
527                            let trimmed = val.trim();
528                            if !trimmed.is_empty() {
529                                doc_comment = Some(trimmed.to_string());
530                            }
531                        }
532                    }
533                }
534            } else if attr.path().is_ident("mro_filename") {
535                if let Meta::NameValue(meta) = &attr.meta {
536                    if let Expr::Lit(elit) = &meta.value {
537                        if let Lit::Str(lstr) = &elit.lit {
538                            mro_filename = Some(lstr.value());
539                        }
540                    }
541                }
542            }
543        }
544        if name.starts_with("__") {
545            return syn::Error::new(
546                field.ident.unwrap().span(),
547                "Identifiers are not allowed to start with __",
548            )
549            .to_compile_error()
550            .into();
551        }
552        if blacklist.contains(name.as_str()) {
553            return syn::Error::new(
554                field.ident.unwrap().span(),
555                format!(
556                    "Field name {} is not allowed here since it is a martian keyword",
557                    name
558                ),
559            )
560            .to_compile_error()
561            .into();
562        }
563        let ty = field.clone().ty;
564
565        let actual_type = match mro_type {
566            Some(t) => quote![#t.parse().unwrap()],
567            None => quote![<#ty as ::martian::AsMartianBlanketType>::as_martian_blanket_type()],
568        };
569
570        if mro_filename.is_some() && doc_comment.is_none() {
571            doc_comment = Some("".to_string());
572        }
573
574        let doc_comment_code = match doc_comment {
575            Some(t) => {
576                quote![Some(#t.to_string())]
577            }
578            None => quote![None],
579        };
580
581        let mro_filename_code = match mro_filename {
582            Some(t) => quote![Some(#t.to_string())],
583            None => quote![None],
584        };
585
586        vec_inner.push(if retain {
587            quote![
588                <::martian::MroField>::retained(#name, #actual_type, #doc_comment_code, #mro_filename_code)
589            ]
590        } else {
591            quote![
592                <::martian::MroField>::new(#name, #actual_type, #doc_comment_code, #mro_filename_code)
593            ]
594        });
595    }
596
597    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
598    // STEP 4
599    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
600    // Generate the `impl MartianStruct` token stream
601    // Handle generics in the struct
602    let (impl_generics, ty_generics, where_clause) = item_struct.generics.split_for_impl();
603    let item_ident = item_struct.ident.clone();
604    let item_ident_str = item_ident.to_string();
605    let final_token = quote![
606        #[automatically_derived]
607        impl #impl_generics ::martian::MartianStruct for #item_ident #ty_generics #where_clause {
608            fn mro_fields() -> Vec<::martian::MroField> {
609                vec![
610                    #(#vec_inner),*
611                ]
612            }
613        }
614
615        #[automatically_derived]
616        impl #impl_generics ::martian::AsMartianPrimaryType for #item_ident #ty_generics #where_clause {
617            fn as_martian_primary_type() -> ::martian::MartianPrimaryType {
618                let fields = <#item_ident #ty_generics as ::martian::MartianStruct>::mro_fields();
619                ::martian::MartianPrimaryType::Struct(
620                    ::martian::mro::StructDef::new(#item_ident_str.into(), fields)
621                )
622            }
623        }
624    ];
625
626    proc_macro::TokenStream::from(final_token)
627}

629/// Custom types which are fields of a `MartianStruct` need to implement `AsMartianBlanketType`.
630/// You can derive that trait on an enum or struct using `#[derive(MartianType)]`
631#[proc_macro_derive(MartianType)]
632pub fn martian_type(item: proc_macro::TokenStream) -> proc_macro::TokenStream {
633    let input = syn::parse_macro_input!(item as DeriveInput);
634    let ident = input.ident.clone();
635    let (impl_generics, ty_generics, where_clause) = input.generics.split_for_impl();
636
637    match input.data {
638        Data::Union(_) => {
639            syn::Error::new_spanned(
640                input,
641                "Usage of union is highly discouraged. MartianType cannot be derived for a Union",
642            )
643            .to_compile_error()
644            .into()
645        },
646        Data::Struct(ref struct_data) => match struct_data.fields {
647            Fields::Unit => {
648                syn::Error::new_spanned(
649                    input,
650                    "MartianType cannot be derived for a unit struct. \
651                    Unit structs don't store any data, so they are most likely not useful as a MartianType.",
652                )
653                .to_compile_error()
654                .into()
655            }
656            Fields::Named(_) => {
657                quote![
658                    #[automatically_derived]
659                    impl #impl_generics ::martian::AsMartianPrimaryType for #ident #ty_generics #where_clause {
660                        fn as_martian_primary_type() -> ::martian::MartianPrimaryType {
661                            ::martian::MartianPrimaryType::Map
662                        }
663                    }
664                ].into()
665            },
666            Fields::Unnamed(_) => {
667                syn::Error::new_spanned(
668                    input,
669                    "Using an tuple struct as an mro field is not recommended. \
670                    The reason is that serde serializes unnamed structs as vectors and it can be represented as a type in martian only if all the fields serialize to the same martian type. \
671                    i.e `struct Good(u8, u16, i32);` can be represented as `int[]`, but there is no martian representation for `struct Bad(u8, String, Foo)`. \
672                    This property is hard to check in a procedural macro. \
673                    Hence it is strongly recommended to use a named struct. \
674                    Naming the fields would also improve the readability of the code.",
675                )
676                .to_compile_error()
677                .into()
678            },
679        },
680        Data::Enum(ref enum_data) => {
681            let mut variant_type_map = HashMap::new();
682            for variant in &enum_data.variants {
683                let this_type = match variant.fields {
684                    Fields::Named(_) => MartianPrimaryType::Map,
685                    Fields::Unnamed(_) => MartianPrimaryType::Map,
686                    Fields::Unit => MartianPrimaryType::Str,
687                };
688                variant_type_map.entry(this_type).or_insert_with(Vec::new).push(variant.ident.to_string());
689            }
690            match variant_type_map.len() {
691                0 => { // Empty Enum
692                    syn::Error::new_spanned(
693                        input,
694                        "MartianType cannot be derived on enums with no variants. They are most likely not useful as a MartianType."
695                    )
696                    .to_compile_error()
697                    .into()
698                },
699                1 => { // All variants mape to either Str or Map, we can generate the derive
700                    if variant_type_map.contains_key(&MartianPrimaryType::Str) {
701                        quote![
702                            #[automatically_derived]
703                            impl #impl_generics ::martian::AsMartianPrimaryType for #ident #ty_generics #where_clause {
704                                fn as_martian_primary_type() -> ::martian::MartianPrimaryType {
705                                    ::martian::MartianPrimaryType::Str
706                                }
707                            }
708                        ].into()
709                    } else {
710                        quote![
711                            #[automatically_derived]
712                            impl #impl_generics ::martian::AsMartianPrimaryType for #ident #ty_generics #where_clause {
713                                fn as_martian_primary_type() -> ::martian::MartianPrimaryType {
714                                    ::martian::MartianPrimaryType::Map
715                                }
716                            }
717                        ].into()
718                    }
719                }
720                2 => {
721                    let map_fields = variant_type_map.get(&MartianPrimaryType::Map).unwrap().join(", ");
722                    let str_fields = variant_type_map.get(&MartianPrimaryType::Str).unwrap().join(", ");
723                    syn::Error::new_spanned(
724                        input,
725                        format!(
726                            "Deriving MartianType on enum {ident} failed because some of the variants in this enum map to MartianPrimaryType::Map while other variants map to MartianPrimaryType::Str.
727  1) MartianPrimaryType::Map -> [{map_fields}]
728  2) MartianPrimaryType::Str -> {str_fields}
729The reason this happens is because serde will deserialize different variants of an enum differently. \
730As a result, we cannot assign a unique martian type for this enum. \
731Consider redesigning your enum to account for this.",
732                        )
733                    )
734                    .to_compile_error()
735                    .into()
736                }
737                _ => unreachable!(),
738            }
739        },
740    }
741}

743/// A macro to define a new struct that implements `MartianFileType` trait
744///
745/// Because this is a procedural macro, as of now, you can only define it
746/// outside functions because they cannot be expanded to statements.
747/// ```rust
748/// use serde::{Serialize, Deserialize};
749/// use martian_derive::martian_filetype;
750/// use martian::MartianFileType;
751/// martian_filetype! { TxtFile, "txt" }
752/// martian_filetype! { BamIndexFile, "bam.bai" }
753///
754/// assert_eq!(TxtFile::extension(), "txt");
755/// assert_eq!(
756///     BamIndexFile::new("/path/to/folder", "filename").as_ref(),
757///     std::path::Path::new("/path/to/folder/filename.bam.bai")
758/// )
759/// ```
760#[proc_macro]
761pub fn martian_filetype(item: proc_macro::TokenStream) -> proc_macro::TokenStream {
762    let item2 = proc_macro2::TokenStream::from(item.clone());
763
764    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
765    // STEP 1
766    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
767    // Convert the tokenstream representing the inputs to the macro as a string.
768    // Do incremental manual parsing instead of a single regex because we want
769    // to generate compile errors that are very specific
770    let input = item.to_string();
771
772    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
773    // STEP 2
774    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
775    // Check that the input is two items separated by a comma. Generate a compile
776    // error if it is not in the expected format. First part is the struct name
777    // and the second part is the extension
778    let parts = input.split(',').collect::<Vec<_>>();
779    if parts.len() != 2 {
780        return syn::Error::new_spanned(
781            item2,
782            "The input to the martian_filetype! macro needs to be two items separated by a comma.
783The first item is the struct name that will be generated and the second item is
784the filetype extension within double quotes.
785For example, martian_filetype! {TxtFile, \"txt\"}",
786        )
787        .to_compile_error()
788        .into();
789    }
790    let struct_name = parts[0].trim();
791    let extension = parts[1].trim();
792
793    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
794    // STEP 3
795    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
796    // Check that the struct name is not empty. Generate a compiler error
797    // otherwise.
798    let struct_ident = match syn::parse_str::<Ident>(struct_name) {
799        Ok(ident) => ident,
800        Err(_) => {
801            return syn::Error::new_spanned(
802                item2,
803                format!(
804                    "The first item `{struct_name}` in the martian_filetype! macro \
805            should be a valid identifier that can be used as a struct name."
806                ),
807            )
808            .to_compile_error()
809            .into()
810        }
811    };
812
813    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
814    // STEP 5
815    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
816    // The struct name is in good shape now. Check that the extension is a
817    // string literal enclosed in double quotes. Generate a compiler error
818    // otherwise.
819    if !(extension.starts_with('"') && extension.ends_with('"')) {
820        return syn::Error::new_spanned(
821            item2,
822            "The second item in the martian_filetype! macro \
823            should be a string literal enclosed in double quotes.",
824        )
825        .to_compile_error()
826        .into();
827    }
828
829    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
830    // STEP 6
831    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
832    // Make sure that the string literal within the quotes in the extension
833    // is not empty. Generate a compiler error otherwise.
834    let chars_within_quotes: Vec<_> = {
835        let mut chars: Vec<_> = extension.chars().skip(1).collect();
836        chars.pop();
837        chars
838    };
839    if chars_within_quotes.is_empty() {
840        return syn::Error::new_spanned(
841            item2,
842            "The extension for a filetype cannot be empty. \
843                Consider using a PathBuf for filenames without any extension.",
844        )
845        .to_compile_error()
846        .into();
847    }
848
849    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
850    // STEP 7
851    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
852    // Make sure that the extension specified but does not start or end with a
853    // dot (.). Generate a compiler error otherwise.
854    if chars_within_quotes[0] == '.' {
855        return syn::Error::new_spanned(
856            item2,
857            "No need to specify the leading dot(.) in the extension",
858        )
859        .to_compile_error()
860        .into();
861    }
862    if *chars_within_quotes.last().unwrap() == '.' {
863        return syn::Error::new_spanned(item2, "Extensions cannot end in a dot(.)")
864            .to_compile_error()
865            .into();
866    }
867
868    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
869    // STEP 8
870    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
871    // Make sure that the extension is ascii alphanumeric or a dot (.). We have
872    // already checked for leading/trailing dots
873    for (i, c) in chars_within_quotes.iter().enumerate() {
874        if !((i > 0 && (c.is_ascii_alphanumeric() || *c == '.')) || c.is_ascii_alphabetic()) {
875            return syn::Error::new_spanned(
876                    item2,
877                    format!(
878                        "The extension `{extension}` in the martian_filetype! macro \
879                        should be alphanumeric (internal dots(.) are okay) starting with an alphabet.\
880                        \nFound invalid character `{c}` at position {i}"))
881                .to_compile_error()
882                .into();
883        }
884    }
885
886    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
887    // STEP 9
888    // ::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::::
889    // Now we are ready to actually generate the code.
890    let extension: String = chars_within_quotes.iter().collect();
891    quote![
892        #[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
893        pub struct #struct_ident(::std::path::PathBuf);
894        #[automatically_derived]
895        impl ::martian::MartianFileType for #struct_ident {
896            /// Returns the extension of this MartianFileType.
897            fn extension() -> String {
898                #extension.into()
899            }
900
901            /// Creates a MartianFileType from a directory path and a filename.
902            fn new(
903                file_path: impl ::std::convert::AsRef<::std::path::Path>,
904                file_name: impl ::std::convert::AsRef<::std::path::Path>,
905            ) -> Self {
906                let path = ::martian::utils::make_path(
907                    file_path.as_ref(),
908                    file_name.as_ref(),
909                    Self::extension(),
910                );
911                #struct_ident(path)
912            }
913        }
914
915        #[automatically_derived]
916        impl ::std::convert::AsRef<::std::path::Path> for #struct_ident {
917            /// Coerces this MartianFileType to a Path slice.
918            fn as_ref(&self) -> &::std::path::Path {
919                &self.0
920            }
921        }
922
923        #[automatically_derived]
924        impl ::std::ops::Deref for #struct_ident {
925            type Target = ::std::path::Path;
926            /// Dereferences this MartianFileType to a Path slice.
927            fn deref(&self) -> &::std::path::Path {
928                &self.0
929            }
930        }
931
932        #[automatically_derived]
933        impl<T> ::std::convert::From<T> for #struct_ident
934        where
935            ::std::path::PathBuf: ::std::convert::From<T>,
936        {
937            /// Convert a PathBuf (or something convertible to a PathBuf) into this MartianFileType.
938            fn from(source: T) -> Self {
939                ::martian::MartianFileType::from_path(::std::path::PathBuf::from(source).as_ref())
940            }
941        }
942    ]
943    .into()
944}
