5// Common parsing for the `.debug_pub*` sections (DWARF v4 Section 6.1.1, Lookup by Name).
6//
7// These sections consist of sets of data. Each set has a header with metadata followed by
8// a series of entries.
9
10#[derive(Clone, Debug)]
11pub(crate) struct DebugPubSet<R: Reader> {
12    pub(crate) section: R,
13}
14
15impl<R: Reader> DebugPubSet<R> {
16    pub(crate) fn sets(&self, is_gnu: bool) -> PubSetIter<R> {
17        PubSetIter {
18            input: self.section.clone(),
19            is_gnu,
20        }
21    }
22
23    pub(crate) fn items(&self, is_gnu: bool) -> PubSetEntryIter<R> {
24        PubSetEntryIter {
25            current_set: None,
26            sets: self.sets(is_gnu),
27        }
28    }
29}

31#[derive(Clone, Debug)]
32pub(crate) struct PubSetIter<R: Reader> {
33    input: R,
34    is_gnu: bool,
35}
36
37impl<R: Reader> PubSetIter<R> {
38    /// Advance the iterator and return the next set.
39    ///
40    /// Returns the newly parsed set as `Ok(Some(PubSet))`. Returns `Ok(None)` when
41    /// iteration is complete. If an error occurs while parsing the next header,
42    /// then this error is returned as `Err(e)`, and all subsequent calls return
43    /// `Ok(None)`.
44    pub(crate) fn next(&mut self) -> Result<Option<PubSet<R>>> {
45        if self.input.is_empty() {
46            return Ok(None);
47        }
48        match PubSetHeader::parse(&mut self.input) {
49            Ok((header, entries)) => Ok(Some(PubSet {
50                header,
51                entries,
52                is_gnu: self.is_gnu,
53            })),
54            Err(e) => {
55                self.input.empty();
56                Err(e)
57            }
58        }
59    }
60}

62#[derive(Clone, Debug)]
63pub(crate) struct PubSetEntryIter<R: Reader> {
64    // Only none at the very beginning and end.
65    // PubSet::entries is consumed as we iterate.
66    current_set: Option<PubSet<R>>,
67    sets: PubSetIter<R>,
68}
69
70impl<R: Reader> PubSetEntryIter<R> {
71    /// Advance the iterator and return the next entry.
72    ///
73    /// Returns the newly parsed entry as `Ok(Some(PubSetEntry))`. Returns
74    /// `Ok(None)` when iteration is complete and all entries have already been
75    /// parsed and yielded. If an error occurs while parsing the next entry,
76    /// then this error is returned as `Err(e)`, and all subsequent calls return
77    /// `Ok(None)`.
78    pub(crate) fn next(&mut self) -> Result<Option<PubSetEntry<R>>> {
79        loop {
80            if let Some(set) = &mut self.current_set
81                && !set.entries.is_empty()
82            {
83                match PubSetEntry::parse(&mut set.entries, &set.header, set.is_gnu) {
84                    Ok(Some(entry)) => return Ok(Some(entry)),
85                    Ok(None) => {
86                        self.current_set = None;
87                    }
88                    Err(e) => {
89                        self.sets.input.empty();
90                        self.current_set = None;
91                        return Err(e);
92                    }
93                }
94            }
95            match self.sets.next() {
96                Ok(Some(set)) => self.current_set = Some(set),
97                Ok(None) => return Ok(None),
98                Err(e) => return Err(e),
99            }
100        }
101    }
102}

106    pub(crate) header: PubSetHeader<R::Offset>,
107    entries: R,
108    is_gnu: bool,
109}

111impl<R: Reader> PubSet<R> {
112    /// Return an iterator over just the entries in this set.
113    pub(crate) fn items(&self) -> PubSetEntryIter<R> {
114        // Empty set iterator.
115        let mut set_input = self.entries.clone();
116        set_input.empty();
117        let sets = PubSetIter {
118            input: set_input,
119            is_gnu: self.is_gnu,
120        };
121
122        PubSetEntryIter {
123            sets,
124            current_set: Some(self.clone()),
125        }
126    }

129#[derive(Debug, Clone)]
130pub(crate) struct PubSetHeader<T = usize> {
131    pub(crate) format: Format,
132    pub(crate) length: T,
133    pub(crate) version: u16,
134    pub(crate) unit_offset: DebugInfoOffset<T>,
135    pub(crate) unit_length: T,
136}
137
138impl<T: ReaderOffset> PubSetHeader<T> {
139    /// Parse a set header. Returns a tuple of the set header and the entry data.
140    fn parse<R: Reader<Offset = T>>(input: &mut R) -> Result<(PubSetHeader<R::Offset>, R)> {
141        let (length, format) = input.read_initial_length()?;
142        let mut rest = input.split(length)?;
143
144        let version = rest.read_u16()?;
145        if version != 2 {
146            return Err(Error::UnknownVersion(u64::from(version)));
147        }
148
149        let unit_offset = rest.read_offset(format).map(DebugInfoOffset)?;
150        let unit_length = rest.read_length(format)?;
151
152        let header = PubSetHeader {
153            format,
154            length,
155            version,
156            unit_offset,
157            unit_length,
158        };
159        Ok((header, rest))
160    }
161}

163#[derive(Debug, Clone)]
164pub(crate) struct PubSetEntry<R: Reader> {
165    pub(crate) unit_header_offset: DebugInfoOffset<R::Offset>,
166    pub(crate) die_offset: UnitOffset<R::Offset>,
167    pub(crate) name: R,
168    flags: u8,
169}
170
171impl<R: Reader> PubSetEntry<R> {
172    pub(crate) fn is_static(&self) -> bool {
173        self.flags & 0x80 != 0
174    }
175
176    pub(crate) fn kind(&self) -> GdbIndexSymbolKind {
177        GdbIndexSymbolKind((self.flags >> 4) & 7)
178    }
179
180    pub(crate) fn flags(&self) -> u8 {
181        self.flags
182    }
183
184    /// Parse a single set entry. Return `None` for the null entry.
185    fn parse(
186        input: &mut R,
187        header: &PubSetHeader<R::Offset>,
188        is_gnu: bool,
189    ) -> Result<Option<PubSetEntry<R>>> {
190        let offset = input.read_offset(header.format)?;
191        if offset.into_u64() == 0 {
192            input.empty();
193            return Ok(None);
194        }
195
196        let flags = if is_gnu { input.read_u8()? } else { 0 };
197        let name = input.read_null_terminated_slice()?;
198        Ok(Some(PubSetEntry {
199            die_offset: UnitOffset(offset),
200            name,
201            flags,
202            unit_header_offset: header.unit_offset,
203        }))
204    }
