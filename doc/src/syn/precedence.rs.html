8// Reference: https://doc.rust-lang.org/reference/expressions.html#expression-precedence
9pub(crate) enum Precedence {
10    // return, break, closures
11    Any,
12    // = += -= *= /= %= &= |= ^= <<= >>=
13    Assign,
14    // .. ..=
15    Range,
16    // ||
17    Or,
18    // &&
19    And,
20    // == != < > <= >=
21    Compare,
22    // |
23    BitOr,
24    // ^
25    BitXor,
26    // &
27    BitAnd,
28    // << >>
29    Shift,
30    // + -
31    Arithmetic,
32    // * / %
33    Term,
34    // as
35    Cast,
36    // unary - * ! & &mut
37    #[cfg(feature = "printing")]
38    Prefix,
39    // function calls, array indexing, field expressions, method calls, ?
40    #[cfg(feature = "printing")]
41    Postfix,
42    // paths, loops
43    #[cfg(feature = "printing")]
44    Unambiguous,
45}
46
47impl Precedence {
48    pub(crate) fn of_binop(op: &BinOp) -> Self {
49        match op {
50            BinOp::Add(_) | BinOp::Sub(_) => Precedence::Arithmetic,
51            BinOp::Mul(_) | BinOp::Div(_) | BinOp::Rem(_) => Precedence::Term,
52            BinOp::And(_) => Precedence::And,
53            BinOp::Or(_) => Precedence::Or,
54            BinOp::BitXor(_) => Precedence::BitXor,
55            BinOp::BitAnd(_) => Precedence::BitAnd,
56            BinOp::BitOr(_) => Precedence::BitOr,
57            BinOp::Shl(_) | BinOp::Shr(_) => Precedence::Shift,
58
59            BinOp::Eq(_)
60            | BinOp::Lt(_)
61            | BinOp::Le(_)
62            | BinOp::Ne(_)
63            | BinOp::Ge(_)
64            | BinOp::Gt(_) => Precedence::Compare,
65
66            BinOp::AddAssign(_)
67            | BinOp::SubAssign(_)
68            | BinOp::MulAssign(_)
69            | BinOp::DivAssign(_)
70            | BinOp::RemAssign(_)
71            | BinOp::BitXorAssign(_)
72            | BinOp::BitAndAssign(_)
73            | BinOp::BitOrAssign(_)
74            | BinOp::ShlAssign(_)
75            | BinOp::ShrAssign(_) => Precedence::Assign,
76        }
77    }
78
79    #[cfg(feature = "printing")]
80    pub(crate) fn of(e: &Expr) -> Self {
81        match e {
82            #[cfg(feature = "full")]
83            Expr::Closure(e) => match e.output {
84                ReturnType::Default => Precedence::Any,
85                ReturnType::Type(..) => Precedence::Unambiguous,
86            },
87
88            Expr::Break(_) | Expr::Return(_) | Expr::Yield(_) => Precedence::Any,
89            Expr::Assign(_) => Precedence::Assign,
90            Expr::Range(_) => Precedence::Range,
91            Expr::Binary(e) => Precedence::of_binop(&e.op),
92            Expr::Cast(_) => Precedence::Cast,
93            Expr::Let(_) | Expr::Reference(_) | Expr::Unary(_) => Precedence::Prefix,
94
95            Expr::Await(_)
96            | Expr::Call(_)
97            | Expr::MethodCall(_)
98            | Expr::Field(_)
99            | Expr::Index(_)
100            | Expr::Try(_) => Precedence::Postfix,
101
102            Expr::Array(_)
103            | Expr::Async(_)
104            | Expr::Block(_)
105            | Expr::Const(_)
106            | Expr::Continue(_)
107            | Expr::ForLoop(_)
108            | Expr::Group(_)
109            | Expr::If(_)
110            | Expr::Infer(_)
111            | Expr::Lit(_)
112            | Expr::Loop(_)
113            | Expr::Macro(_)
114            | Expr::Match(_)
115            | Expr::Paren(_)
116            | Expr::Path(_)
117            | Expr::Repeat(_)
118            | Expr::Struct(_)
119            | Expr::TryBlock(_)
120            | Expr::Tuple(_)
121            | Expr::Unsafe(_)
122            | Expr::Verbatim(_)
123            | Expr::While(_) => Precedence::Unambiguous,
124
125            #[cfg(not(feature = "full"))]
126            Expr::Closure(_) => unreachable!(),
127        }
128    }
129
130    #[cfg(feature = "printing")]
131    pub(crate) fn of_rhs(e: &Expr) -> Self {
132        match e {
133            Expr::Break(_) | Expr::Closure(_) | Expr::Return(_) | Expr::Yield(_) => {
134                Precedence::Prefix
135            }
136            #[cfg(feature = "full")]
137            Expr::Range(e) if e.start.is_none() => Precedence::Prefix,
138            _ => Precedence::of(e),
139        }
140    }
141}

159        let this = *self as u8;
160        let other = *other as u8;
161        Some(this.cmp(&other))
162    }
