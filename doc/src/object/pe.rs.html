17/// Platform-specific constant names for a PE/COFF file.
18///
19/// Returned by [`names`] and [`machine_names`].
20#[cfg(feature = "names")]
21#[derive(Debug)]
22#[non_exhaustive]
23pub struct Names {
24    /// Values for `ImageRelocation::typ`.
25    pub rel: &'static FlagNames<RelocationType>,
26    /// Values for the type in an `ImageBaseRelocation` block.
27    pub rel_based: &'static ConstantNames<BaseRelocationType>,
28}
29
30/// Return the platform independent names for constants.
31#[cfg(feature = "names")]
32pub const fn names() -> &'static Names {
33    Base::names()
34}
35
36/// Return the platform specific names for constants.
37///
38/// Note that these also include the values returned by [`names`].
39#[cfg(feature = "names")]
40pub const fn machine_names(machine: Machine) -> &'static Names {
41    match machine {
42        IMAGE_FILE_MACHINE_I386 => I386::names(),
43        IMAGE_FILE_MACHINE_MIPS16 | IMAGE_FILE_MACHINE_MIPSFPU | IMAGE_FILE_MACHINE_MIPSFPU16 => {
44            Mips::names()
45        }
46        IMAGE_FILE_MACHINE_ALPHA | IMAGE_FILE_MACHINE_ALPHA64 => Alpha::names(),
47        IMAGE_FILE_MACHINE_POWERPC
48        | IMAGE_FILE_MACHINE_POWERPCFP
49        | IMAGE_FILE_MACHINE_POWERPCBE => Ppc::names(),
50        IMAGE_FILE_MACHINE_SH3
51        | IMAGE_FILE_MACHINE_SH3DSP
52        | IMAGE_FILE_MACHINE_SH3E
53        | IMAGE_FILE_MACHINE_SH4
54        | IMAGE_FILE_MACHINE_SH5 => Sh::names(),
55        IMAGE_FILE_MACHINE_ARM => Arm::names(),
56        IMAGE_FILE_MACHINE_AM33 => Am::names(),
57        IMAGE_FILE_MACHINE_ARM64 => Arm64::names(),
58        IMAGE_FILE_MACHINE_AMD64 => Amd64::names(),
59        IMAGE_FILE_MACHINE_IA64 => Ia64::names(),
60        IMAGE_FILE_MACHINE_CEF => Cef::names(),
61        IMAGE_FILE_MACHINE_CEE => Cee::names(),
62        IMAGE_FILE_MACHINE_M32R => M32r::names(),
63        IMAGE_FILE_MACHINE_EBC => Ebc::names(),
64        IMAGE_FILE_MACHINE_RISCV32 | IMAGE_FILE_MACHINE_RISCV64 | IMAGE_FILE_MACHINE_RISCV128 => {
65            Riscv::names()
66        }
67        _ => Base::names(),
68    }
69}

304/// A PE rich header entry.
305///
306/// Rich headers have no official documentation, but have been heavily
307/// reversed-engineered and documented in the wild, e.g.:

309/// * `https://www.researchgate.net/figure/Structure-of-the-Rich-Header_fig1_318145388`
310///
311/// This data is "masked", i.e. XORed with a checksum derived from the file data.

2414//
2415
2416//
2417// Resource directory consists of two counts, following by a variable length
2418// array of directory entries.  The first count is the number of entries at
2419// beginning of the array that have actual names associated with each entry.
2420// The entries are in ascending order, case insensitive strings.  The second
2421// count is the number of entries that immediately follow the named entries.
2422// This second count identifies the number of entries that have 16-bit integer
2423// Ids as their name.  These entries are also sorted in ascending order.
2424//
2425// This structure allows fast lookup by either name or number, but for any
2426// given resource entry only one form of lookup is supported, not both.
2427// This is consistent with the syntax of the .RC file and the .RES file.
2428//
2429

2437    pub number_of_named_entries: U16<LE>,
2438    pub number_of_id_entries: U16<LE>,
2439}

2441pub const IMAGE_RESOURCE_NAME_IS_STRING: u32 = 0x8000_0000;
2442pub const IMAGE_RESOURCE_DATA_IS_DIRECTORY: u32 = 0x8000_0000;
2443//
2444// Each directory contains the 32-bit Name of the entry and an offset,
2445// relative to the beginning of the resource directory of the data associated
2446// with this directory entry.  If the name of the entry is an actual text
2447// string instead of an integer Id, then the high order bit of the name field
2448// is set to one and the low order 31-bits are an offset, relative to the
2449// beginning of the resource directory of the string, which is of type
2450// IMAGE_RESOURCE_DIRECTORY_STRING.  Otherwise the high bit is clear and the
2451// low-order 16-bits are the integer Id that identify this resource directory
2452// entry. If the directory entry is yet another resource directory (i.e. a
2453// subdirectory), then the high order bit of the offset field will be
2454// set to indicate this.  Otherwise the high bit is clear and the offset
2455// field points to a resource data entry.
2456//
