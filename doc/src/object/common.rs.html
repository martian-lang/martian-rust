320/// The operation used to calculate the result of the relocation.
321///
322/// The relocation descriptions use the following definitions. Note that
323/// these definitions probably don't match any ELF ABI.
324///
325/// * A - The value of the addend.
326/// * G - The address of the symbol's entry within the global offset table.
327/// * L - The address of the symbol's entry within the procedure linkage table.
328/// * P - The address of the place of the relocation.
329/// * S - The address of the symbol.
330/// * GotBase - The address of the global offset table.
331/// * Image - The base address of the image.
332/// * Section - The address of the section containing the symbol.
333///
334/// 'XxxRelative' means 'Xxx + A - P'.  'XxxOffset' means 'S + A - Xxx'.
335#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
336#[non_exhaustive]
337pub enum RelocationKind {
338    /// The operation is unknown.
339    Unknown,
340    /// No relocation.
341    ///
342    /// Example: `elf::R_X86_64_NONE`
343    None,
344    /// S + A
345    ///
346    /// Example: `elf::R_X86_64_64`
347    Absolute,
348    /// S + A - P
349    ///
350    /// Example: `elf::R_X86_64_PC32`
351    Relative,
352    /// G + A - GotBase
353    ///
354    /// Example: `elf::R_X86_64_GOT32`
355    Got,
356    /// G + A - P
357    ///
358    /// Example: `elf::R_X86_64_GOTPCREL`
359    GotRelative,
360    /// GotBase + A - P
361    ///
362    /// Example: `elf::R_386_GOTPC`
363    GotBaseRelative,
364    /// S + A - GotBase
365    ///
366    /// Example: `elf::R_386_GOTOFF`
367    GotBaseOffset,
368    /// L + A - P
369    ///
370    /// Example: `elf::R_X86_64_PLT32`
371    PltRelative,
372    /// S + A - Image
373    ///
374    /// Example: `pe::IMAGE_REL_AMD64_ADDR32NB`
375    ImageOffset,
376    /// S + A - Section
377    ///
378    /// Example: `pe::IMAGE_REL_AMD64_SECREL`
379    SectionOffset,
380    /// The index of the section containing the symbol.
381    ///
382    /// Example: `pe::IMAGE_REL_AMD64_SECTION`
383    SectionIndex,
384}

536/// Memory permissions for a segment.
537///
538/// This is a simplified representation of segment permissions that abstracts
539/// over format-specific flags.
540#[derive(Clone, Copy, PartialEq, Eq, Hash, Default)]
541pub struct Permissions {
542    bits: u8,
543}
544
545impl core::fmt::Debug for Permissions {
546    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
547        write!(
548            f,
549            "{}{}{}",
550            if self.readable() { 'R' } else { '-' },
551            if self.writable() { 'W' } else { '-' },
552            if self.executable() { 'X' } else { '-' },
553        )
554    }
555}
556
557impl Permissions {
558    /// Permission bit for readable.
559    const R: u8 = 1 << 0;
560    /// Permission bit for writable.
561    const W: u8 = 1 << 1;
562    /// Permission bit for executable.
563    const X: u8 = 1 << 2;
564
565    /// Creates a new `Permissions` with the given flags.
566    pub fn new(readable: bool, writable: bool, executable: bool) -> Self {
567        let mut bits = 0;
568        if readable {
569            bits |= Self::R;
570        }
571        if writable {
572            bits |= Self::W;
573        }
574        if executable {
575            bits |= Self::X;
576        }
577        Permissions { bits }
578    }
579
580    /// Returns true if the segment is readable.
581    #[inline]
582    pub fn readable(&self) -> bool {
583        self.bits & Self::R != 0
584    }
585
586    /// Returns true if the segment is writable.
587    #[inline]
588    pub fn writable(&self) -> bool {
589        self.bits & Self::W != 0
590    }
591
592    /// Returns true if the segment is executable.
593    #[inline]
594    pub fn executable(&self) -> bool {
595        self.bits & Self::X != 0
596    }
597
598    /// Returns true if the segment is readable but not writable.
599    #[inline]
600    pub fn readonly(&self) -> bool {
601        self.readable() && !self.writable()
602    }
603}

727pub enum RelocationFlags {
728    /// Format independent representation.
729    Generic {
730        /// The operation used to calculate the result of the relocation.
731        kind: RelocationKind,
732        /// Information about how the result of the relocation operation is encoded in the place.
733        encoding: RelocationEncoding,
734        /// The size in bits of the place of relocation.
735        size: u8,
736    },
737    /// ELF relocation fields.
738    #[cfg(feature = "elf")]
739    Elf {
740        /// `r_type` field in the ELF relocation.
741        r_type: crate::elf::RelocationType,
742    },
743    /// Mach-O relocation fields.
744    #[cfg(feature = "macho")]
745    MachO {
746        /// `r_type` field in the Mach-O relocation.
747        r_type: crate::macho::RelocationType,
748        /// `r_pcrel` field in the Mach-O relocation.
749        r_pcrel: bool,
750        /// `r_length` field in the Mach-O relocation.
751        r_length: u8,
752    },
753    /// COFF relocation fields.
754    #[cfg(feature = "coff")]
755    Coff {
756        /// `typ` field in the COFF relocation.
757        typ: crate::pe::RelocationType,
758    },
759    /// XCOFF relocation fields.
760    #[cfg(feature = "xcoff")]
761    Xcoff {
762        /// `r_rtype` field in the XCOFF relocation.
763        r_rtype: crate::xcoff::RelocationType,
764        /// `r_rsize` field in the XCOFF relocation.
765        r_rsize: u8,
766    },
767    /// Wasm relocation fields.
768    #[cfg(feature = "wasm")]
769    Wasm {
770        /// Relocation type (the `R_WASM_*` constant).
771        r_type: u8,
772    },
773}

775/// Wrapper to print as `[..]` without a manual `Debug` implementation, rather than dumping an
776/// entire byte array.
777#[allow(unused)]
778#[derive(Clone, Copy, Hash, PartialEq, Eq, PartialOrd, Ord)]
779pub(crate) struct SkipDebugList<T>(pub T);
780
781impl<T> fmt::Debug for SkipDebugList<T> {
782    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
783        write!(f, "[..]")
784    }
785}
786
787impl<T> Deref for SkipDebugList<T> {
788    type Target = T;
789
790    fn deref(&self) -> &Self::Target {
791        &self.0
792    }
793}
794
795impl<T> DerefMut for SkipDebugList<T> {
796    fn deref_mut(&mut self) -> &mut Self::Target {
797        &mut self.0
798    }
799}
