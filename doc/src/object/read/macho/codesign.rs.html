5impl<E: Endian> macho::LinkeditDataCommand<E> {
6    /// Parse the data referenced by an `LC_CODE_SIGNATURE` load command.
7    pub fn code_signature<'data, R: ReadRef<'data>>(
8        &self,
9        endian: E,
10        data: R,
11    ) -> Result<CodeSignature<'data>> {
12        let signature_data = data
13            .read_bytes_at(
14                self.dataoff.get(endian).into(),
15                self.datasize.get(endian).into(),
16            )
17            .read_error("Invalid Mach-O code signature offset or size")?;
18        CodeSignature::parse(signature_data)
19    }
20}

22/// The parsed data of an `LC_CODE_SIGNATURE` load command.
23///
24/// This is a [`macho::CsSuperBlob`] header and an index table for the blobs.
25#[derive(Debug, Clone, Copy)]
26pub struct CodeSignature<'data> {
27    data: Bytes<'data>,
28    header: &'data macho::CsSuperBlob,
29    index: &'data [macho::CsBlobIndex],
30}
31
32impl<'data> CodeSignature<'data> {
33    /// Parse the code signature data.
34    ///
35    /// `data` should be the data referenced by the `LC_CODE_SIGNATURE` load command.
36    pub fn parse(data: &'data [u8]) -> Result<Self> {
37        let mut bytes = Bytes(data);
38        let header = bytes
39            .read::<macho::CsSuperBlob>()
40            .read_error("Invalid Mach-O code signature super blob")?;
41        if header.magic.get(BigEndian) != macho::CSMAGIC_EMBEDDED_SIGNATURE {
42            return Err(Error("Unsupported Mach-O code signature magic"));
43        }
44        let index = bytes
45            .read_slice::<macho::CsBlobIndex>(header.count.get(BigEndian) as usize)
46            .read_error("Invalid Mach-O code signature blob count")?;
47        Ok(CodeSignature {
48            data: Bytes(data),
49            header,
50            index,
51        })
52    }
53
54    /// Return the super blob header.
55    pub fn header(&self) -> &'data macho::CsSuperBlob {
56        self.header
57    }
58
59    /// Return the blob index.
60    pub fn index(&self) -> &'data [macho::CsBlobIndex] {
61        self.index
62    }
63
64    /// Return an iterator over the blobs in the code signature.
65    pub fn blobs(&self) -> CodeSignatureBlobIterator<'data> {
66        CodeSignatureBlobIterator {
67            data: self.data,
68            index: self.index.iter(),
69        }
70    }

73/// An iterator over the blobs in a [`CodeSignature`].
74///
75/// Returned by [`CodeSignature::blobs`].
76#[derive(Debug, Clone)]
77pub struct CodeSignatureBlobIterator<'data> {
78    data: Bytes<'data>,
79    index: core::slice::Iter<'data, macho::CsBlobIndex>,
80}

82impl<'data> CodeSignatureBlobIterator<'data> {
83    /// Return the next blob.
84    pub fn next(&mut self) -> Result<Option<CodeSignatureBlob<'data>>> {
85        let Some(index) = self.index.next() else {
86            return Ok(None);
87        };
88        let slot = index.slot.get(BigEndian);
89        let offset = index.offset.get(BigEndian);
90        let header = self
91            .data
92            .read_at::<macho::CsGenericBlob>(offset as usize)
93            .read_error("Invalid Mach-O code signature blob offset")?;
94        let data = self
95            .data
96            .read_bytes_at(offset as usize, header.length.get(BigEndian) as usize)
97            .read_error("Invalid Mach-O code signature blob length")?;
98        Ok(Some(CodeSignatureBlob {
99            slot,
100            offset,
101            magic: header.magic.get(BigEndian),
102            data,
103        }))
104    }
105}

172pub struct CodeDirectory<'data> {
173    data: Bytes<'data>,
174    header: &'data macho::CsCodeDirectoryV0,
175    v1: Option<&'data macho::CsCodeDirectoryV1>,
176    v2: Option<&'data macho::CsCodeDirectoryV2>,
177    v3: Option<&'data macho::CsCodeDirectoryV3>,
178    v4: Option<&'data macho::CsCodeDirectoryV4>,
179}

181impl<'data> CodeDirectory<'data> {
182    /// Parse a code directory blob.
183    ///
184    /// `data` should be the entire blob, including the magic number and length.
185    pub fn parse(data: &'data [u8]) -> Result<Self> {
186        let mut header_data = Bytes(data);
187        let header = header_data
188            .read::<macho::CsCodeDirectoryV0>()
189            .read_error("Invalid Mach-O code directory")?;
190        if header.magic.get(BigEndian) != macho::CSMAGIC_CODEDIRECTORY {
191            return Err(Error("Unsupported Mach-O code directory magic"));
192        }
193        let version = header.version.get(BigEndian);
194
195        let v1 = if version >= macho::CS_SUPPORTSSCATTER {
196            Some(
197                header_data
198                    .read::<macho::CsCodeDirectoryV1>()
199                    .read_error("Invalid Mach-O code directory length")?,
200            )
201        } else {
202            None
203        };
204        let v2 = if version >= macho::CS_SUPPORTSTEAMID {
205            Some(
206                header_data
207                    .read::<macho::CsCodeDirectoryV2>()
208                    .read_error("Invalid Mach-O code directory length")?,
209            )
210        } else {
211            None
212        };
213        let v3 = if version >= macho::CS_SUPPORTSCODELIMIT64 {
214            Some(
215                header_data
216                    .read::<macho::CsCodeDirectoryV3>()
217                    .read_error("Invalid Mach-O code directory length")?,
218            )
219        } else {
220            None
221        };
222        let v4 = if version >= macho::CS_SUPPORTSEXECSEG {
223            Some(
224                header_data
225                    .read::<macho::CsCodeDirectoryV4>()
226                    .read_error("Invalid Mach-O code directory length")?,
227            )
228        } else {
229            None
230        };
231
232        Ok(CodeDirectory {
233            data: Bytes(data),
234            header,
235            v1,
236            v2,
237            v3,
238            v4,
239        })
240    }

300    /// Return the team identifier string, if present.
301    pub fn team_id(&self) -> Result<Option<&'data [u8]>> {
302        let team_offset = self.team_offset().unwrap_or(0);
303        if team_offset == 0 {
304            return Ok(None);
305        }
306        let team_id = self
307            .data
308            .read_string_at(team_offset as usize)

310        Ok(Some(team_id))

312
313    /// Return the hash for a special slot.
314    ///
315    /// Special slots are identified by a [`macho::CsSlot`] value in the range
316    /// `1..=header.n_special_slots`.
317    ///
318    /// Returns an error if slot is not present.
319    pub fn special_hash(&self, slot: macho::CsSlot) -> Result<&'data [u8]> {
320        if slot.0 < 1 || slot.0 > self.header.n_special_slots.get(BigEndian) {
321            return Err(Error("Invalid Mach-O code directory special hash index"));
322        }
323        let offset = slot
324            .0
325            .checked_mul(u32::from(self.header.hash_size))
326            .and_then(|offset| self.header.hash_offset.get(BigEndian).checked_sub(offset))
327            .read_error("Invalid Mach-O code directory hash size")?;
328        Ok(self
329            .data
330            .read_bytes_at(offset as usize, self.header.hash_size as usize)
331            .read_error("Invalid Mach-O code directory hash offset")?
332            .0)
333    }
334
335    /// Return the code hash for the given page index.
336    ///
337    /// Returns an error if index is greater than or equal to `n_code_slots`.
338    pub fn code_hash(&self, index: u32) -> Result<&'data [u8]> {
339        if self.scatter_offset().unwrap_or(0) != 0 {
340            return Err(Error("Unsupported Mach-O code directory scatter"));
341        }
342        if index >= self.header.n_code_slots.get(BigEndian) {
343            return Err(Error("Invalid Mach-O code directory hash index"));
344        }
345        let offset = index
346            .checked_mul(u32::from(self.header.hash_size))
347            .and_then(|offset| offset.checked_add(self.header.hash_offset.get(BigEndian)))
348            .read_error("Invalid Mach-O code directory hash size")?;
349        Ok(self
350            .data
351            .read_bytes_at(offset as usize, self.header.hash_size as usize)
352            .read_error("Invalid Mach-O code directory hash offset")?
353            .0)
354    }
