10/// An iterator for the load commands from a [`MachHeader`].
11#[derive(Debug, Default, Clone, Copy)]
12pub struct LoadCommandIterator<'data, E: Endian> {
13    endian: E,
14    data: Bytes<'data>,
15    ncmds: u32,
16    offset: u64,
17}
18
19impl<'data, E: Endian> LoadCommandIterator<'data, E> {
20    pub(super) fn new(endian: E, data: &'data [u8], ncmds: u32, offset: u64) -> Self {
21        LoadCommandIterator {
22            endian,
23            data: Bytes(data),
24            ncmds,
25            offset,
26        }
27    }
28
29    /// Return the next load command.
30    pub fn next(&mut self) -> Result<Option<LoadCommandData<'data, E>>> {
31        if self.ncmds == 0 {
32            return Ok(None);
33        }
34
35        let result = self.parse().map(Some);
36        if result.is_err() {
37            self.ncmds = 0;
38        } else {
39            self.ncmds -= 1;
40        }
41        result
42    }
43
44    fn parse(&mut self) -> Result<LoadCommandData<'data, E>> {
45        let header = self
46            .data
47            .read_at::<macho::LoadCommand<E>>(0)
48            .read_error("Invalid Mach-O load command header")?;
49        let cmd = header.cmd.get(self.endian);
50        let cmdsize = header.cmdsize.get(self.endian) as usize;
51        if cmdsize < mem::size_of::<macho::LoadCommand<E>>() {
52            return Err(Error("Invalid Mach-O load command size"));
53        }
54        let data = self
55            .data
56            .read_bytes(cmdsize)
57            .read_error("Invalid Mach-O load command size")?;
58        let offset = self.offset;
59        self.offset += cmdsize as u64;
60        Ok(LoadCommandData {
61            cmd,
62            offset,
63            data,
64            marker: Default::default(),
65        })
66    }
67}

69impl<'data, E: Endian> Iterator for LoadCommandIterator<'data, E> {
70    type Item = Result<LoadCommandData<'data, E>>;
71
72    fn next(&mut self) -> Option<Self::Item> {
73        self.next().transpose()
74    }
75}

247    /// Parse the optional flags field for a dylib load command.
248    ///
249    /// [`macho::DylibCommand`] traditionally uses the load command type to distinguish
250    /// between dylib kinds. [`macho::DylibUseCommand`] replaces this by encoding the
251    /// kinds in a bitfield appended after the standard `DylibCommand` fields. Its
252    /// presence is signalled using sentinel values in some `DylibCommand` fields.
253    ///
254    /// Returns `None` if the sentinels are absent. If `Some` is returned, the value of
255    /// [`macho::Dylib::timestamp`] should be ignored.
256    pub fn dylib_use_flags(
257        self,
258        endian: E,
259        dylib: &macho::DylibCommand<E>,
260    ) -> Result<Option<macho::DylibUseFlags>> {
261        if dylib.dylib.name.offset.get(endian) != 28 // size of DylibUseCommand
262            || dylib.dylib.timestamp.get(endian) != macho::DYLIB_USE_MARKER
263        {
264            return Ok(None);
265        }
266        Ok(Some(
267            self.data
268                .read_at::<U32<_, _>>(24) // offset of DylibUseCommand::flags
269                .read_error("Invalid dylib load command size")?
270                .get(endian),
271        ))
272    }

454impl<E: Endian> macho::SymtabCommand<E> {
455    /// Return the symbol table that this command references.
456    pub fn symbols<'data, Mach: MachHeader<Endian = E>, R: ReadRef<'data>>(
457        &self,
458        endian: E,
459        data: R,
460    ) -> Result<SymbolTable<'data, Mach, R>> {
461        let symbols = data
462            .read_slice_at(
463                self.symoff.get(endian).into(),
464                self.nsyms.get(endian) as usize,
465            )
466            .read_error("Invalid Mach-O symbol table offset or size")?;
467        let str_start: u64 = self.stroff.get(endian).into();
468        let str_end = str_start
469            .checked_add(self.strsize.get(endian).into())
470            .read_error("Invalid Mach-O string table length")?;
471        let strings = StringTable::new(data, str_start, str_end);
472        Ok(SymbolTable::new(symbols, strings))
473    }
474}

496impl<E: Endian> macho::LinkeditDataCommand<E> {
497    /// Return the linkedit data referenced by the command.
498    pub fn data<'data, R: ReadRef<'data>>(&self, endian: E, data: R) -> Result<&'data [u8]> {
499        data.read_bytes_at(
500            self.dataoff.get(endian).into(),
501            self.datasize.get(endian).into(),
502        )
503        .read_error("Invalid linkedit offset or size")
504    }
505
506    /// Return an iterator over the function start addresses.
507    ///
508    /// Only works if the command is a `LC_FUNCTION_STARTS` command.
509    ///
510    /// # Arguments
511    /// * `text_segment_addr` - The VM address of the __TEXT segment.
512    pub fn function_starts<'data, R: ReadRef<'data>>(
513        &self,
514        endian: E,
515        data: R,
516        text_segment_addr: u64,
517    ) -> Result<FunctionStartsIterator<'data>> {
518        if self.cmd.get(endian) != macho::LC_FUNCTION_STARTS {
519            return Err(Error("Not a function starts command"));
520        }
521        let data = self.data(endian, data)?;
522        Ok(FunctionStartsIterator::new(data, text_segment_addr))
523    }
524
525    /// Return an iterator over the exports trie.
526    ///
527    /// Only works if the command is a `LC_DYLD_EXPORTS_TRIE` command.
528    pub fn exports_trie<'data, R: ReadRef<'data>>(
529        &self,
530        endian: E,
531        data: R,
532    ) -> Result<ExportsTrieIterator<'data>> {
533        if self.cmd.get(endian) != macho::LC_DYLD_EXPORTS_TRIE {
534            return Err(Error("Not an exports trie command"));
535        }
536        let data = self.data(endian, data)?;
537        Ok(ExportsTrieIterator::new(data))
538    }
