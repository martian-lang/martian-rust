22/// An iterator for the exports in a [`MachOFile`].
23pub struct MachOExportIterator<'data, 'file, Mach, R = &'data [u8]>
24where
25    Mach: MachHeader,
26    R: ReadRef<'data>,
27{
28    internal: MachOExportIteratorInternal<'data, 'file, Mach, R>,
29}

31enum MachOExportIteratorInternal<'data, 'file, Mach, R>
32where
33    Mach: MachHeader,
34    R: ReadRef<'data>,
35{
36    Symbols {
37        file: &'file MachOFile<'data, Mach, R>,

41        file: &'file MachOFile<'data, Mach, R>,

58impl<'data, 'file, Mach, R> MachOExportIterator<'data, 'file, Mach, R>
59where
60    Mach: MachHeader,
61    R: ReadRef<'data>,
62{
63    pub(super) fn new(file: &'file MachOFile<'data, Mach, R>) -> Result<Self> {
64        let mut exports_trie = None;
65        let mut dyld_info = None;
66        let mut commands = file.macho_load_commands()?;
67        while let Some(command) = commands.next()? {
68            if let Some(command) = command.dyld_exports_trie()? {
69                exports_trie = Some(command);
70            } else if let Some(command) = command.dyld_info()? {
71                dyld_info = Some(command);
72            }
73        }
74        let trie_iter = if let Some(exports_trie) = exports_trie {
75            let data = file
76                .linkedit_data
77                .read_error("Missing Mach-O linkedit segment")?;
78            Some(exports_trie.exports_trie(file.endian, data.0)?)
79        } else if let Some(dyld_info) = dyld_info {
80            let data = file
81                .linkedit_data
82                .read_error("Missing Mach-O linkedit segment")?;
83            Some(dyld_info.exports_trie(file.endian, data.0)?)
84        } else {
85            None
86        };
87        let internal = if let Some(iter) = trie_iter {
88            let image_base = file
89                .segments
90                .iter()
91                .find(|s| s.segment.name() == macho::SEG_TEXT.as_bytes())
92                .map(|s| s.segment.vmaddr(file.endian).into())
93                .unwrap_or(0);
94            MachOExportIteratorInternal::ExportsTrie {
95                file,
96                image_base,
97                libraries: None,
98                iter,
99            }
100        } else {
101            // `LC_DYSYMTAB` is not required, so use the symbol table directly.
102            // This also allows us to find `N_INDR` symbols.
103            let iter = file.symbols.symbols().iter();
104            MachOExportIteratorInternal::Symbols { file, iter }
105        };
106        Ok(MachOExportIterator { internal })
107    }
108
109    fn next(&mut self) -> Result<Option<Export<'data>>> {
110        match &mut self.internal {
111            MachOExportIteratorInternal::ExportsTrie {
112                file,
113                image_base,
114                libraries,
115                iter,
116            } => {
117                let Some(symbol) = iter.next()? else {
118                    return Ok(None);
119                };
120                let flags = symbol.flags();
121                let target = match symbol.data() {
122                    ExportData::Regular { address } => match flags.kind() {
123                        macho::EXPORT_SYMBOL_FLAGS_KIND_ABSOLUTE => {
124                            ExportTarget::Absolute { value: *address }
125                        }
126                        macho::EXPORT_SYMBOL_FLAGS_KIND_THREAD_LOCAL => {
127                            ExportTarget::TlvDescriptor {
128                                address: image_base.wrapping_add(*address),
129                            }
130                        }
131                        macho::EXPORT_SYMBOL_FLAGS_KIND_REGULAR => ExportTarget::Address {
132                            address: image_base.wrapping_add(*address),
133                        },
134                        _ => {
135                            return Err(Error("Unsupported Mach-O export kind"));
136                        }
137                    },
138                    ExportData::StubAndResolver {
139                        stub_address,
140                        resolver_address,
141                    } => {
142                        if flags.kind() != macho::EXPORT_SYMBOL_FLAGS_KIND_REGULAR {
143                            return Err(Error("Unsupported Mach-O export kind"));
144                        }
145                        ExportTarget::Resolver {
146                            resolver: image_base.wrapping_add(*resolver_address),
147                            stub: Some(image_base.wrapping_add(*stub_address)),
148                        }
149                    }
150                    ExportData::Reexport {
151                        dylib_ordinal,
152                        import_name,
153                    } => {
154                        if libraries.is_none() {
155                            *libraries = Some(file.libraries()?);
156                        }
157                        let library = usize::try_from(*dylib_ordinal)
158                            .ok()
159                            // No special ordinals.
160                            .filter(|index| *index != 0)
161                            .and_then(|index| libraries.as_deref()?.get(index).copied())
162                            .read_error("Invalid Mach-O export dylib ordinal")?;
163                        ExportTarget::Reexport {
164                            library,
165                            name: NameOrOrdinal::Name(import_name),
166                        }
167                    }
168                };
169                Ok(Some(Export {
170                    name: NameOrOrdinal::Name(Cow::Owned(symbol.into_name())),
171                    target,
172                    weak: flags.contains(macho::EXPORT_SYMBOL_FLAGS_WEAK_DEFINITION),
173                    flags: ExportFlags::None,
174                }))
175            }
176            MachOExportIteratorInternal::Symbols { file, iter } => {
177                let endian = file.endian;
178                let symbols = &file.symbols;
179                loop {
180                    let Some(symbol) = iter.next() else {
181                        return Ok(None);
182                    };
183                    // The above iterator has made progress, so errors after here don't need to
184                    // terminate iteration.
185
186                    let n_type = symbol.n_type();
187                    if n_type.is_stab() || !n_type.is_ext() || n_type.is_pext() {
188                        continue;
189                    }
190                    let name = symbol.name(endian, symbols.strings())?;
191                    let value = symbol.n_value(endian).into();
192                    let target = match symbol.n_type().typ() {
193                        macho::N_ABS => ExportTarget::Absolute { value },
194                        macho::N_INDR => {
195                            let import_name = u32::try_from(value)
196                                .ok()
197                                .and_then(|offset| symbols.strings().get(offset).ok())
198                                .read_error("Invalid Mach-O indirect symbol value")?;
199                            ExportTarget::Reexport {
200                                library: &[],
201                                name: NameOrOrdinal::Name(import_name),
202                            }
203                        }
204                        macho::N_SECT => {
205                            if symbol.n_desc(endian).contains(macho::N_SYMBOL_RESOLVER) {
206                                ExportTarget::Resolver {
207                                    resolver: value,
208                                    stub: None,
209                                }
210                            } else {
211                                let index = SectionIndex(symbol.n_sect().into());
212                                let section = file.section_internal(index)?;
213                                if section.section.flags(endian).typ()
214                                    == macho::S_THREAD_LOCAL_VARIABLES
215                                {
216                                    ExportTarget::TlvDescriptor { address: value }
217                                } else {
218                                    ExportTarget::Address { address: value }
219                                }
220                            }
221                        }
222                        _ => continue,
223                    };
224                    break Ok(Some(Export {
225                        name: NameOrOrdinal::Name(Cow::Borrowed(name)),
226                        target,
227                        weak: symbol.n_desc(endian).contains(macho::N_WEAK_DEF),
228                        flags: ExportFlags::MachO {
229                            n_type: symbol.n_type(),
230                            n_desc: symbol.n_desc(endian),
231                        },
232                    }));
233                }
234            }
235        }
236    }
237}
