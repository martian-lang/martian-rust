8    /// Parse the data referenced by an `LC_DYLD_CHAINED_FIXUPS` load command.

21    }

24/// The parsed data of an `LC_DYLD_CHAINED_FIXUPS` load command.
25///
26/// This gives access to the [`macho::DyldChainedFixupsHeader`], the imports table,
27/// and the per-segment chain starts. The fixup chains themselves are located in the
28/// segment data, and can be walked using [`DyldChainedSegment::fixups`].
29#[derive(Debug, Clone, Copy)]
30pub struct DyldChainedFixups<'data, E: Endian> {
31    data: Bytes<'data>,
32    header: &'data macho::DyldChainedFixupsHeader<E>,
33}
34
35impl<'data, E: Endian> DyldChainedFixups<'data, E> {
36    /// Parse the chained fixups data.
37    ///
38    /// `data` should be the data referenced by the `LC_DYLD_CHAINED_FIXUPS` load command.
39    pub fn parse(endian: E, data: &'data [u8]) -> Result<Self> {
40        let header = Bytes(data)
41            .read_at::<macho::DyldChainedFixupsHeader<E>>(0)
42            .read_error("Invalid Mach-O chained fixups header")?;
43        if header.fixups_version.get(endian) != 0 {
44            return Err(Error("Unsupported Mach-O chained fixups version"));
45        }
46        Ok(DyldChainedFixups {
47            data: Bytes(data),
48            header,
49        })
50    }
51
52    /// Return the chained fixups header.
53    pub fn header(&self) -> &'data macho::DyldChainedFixupsHeader<E> {
54        self.header
55    }
56
57    /// Return an iterator over the imports.
58    pub fn imports(&self, endian: E) -> Result<DyldChainedImportIterator<'data, E>> {
59        if self.header.symbols_format.get(endian) != 0 {
60            return Err(Error(
61                "Unsupported Mach-O chained fixups compressed symbols",
62            ));
63        }
64        // The imports table and the symbol string pool each extend to the end of the
65        // chain data.
66        let data = self.data.0;
67        let imports = data
68            .get(self.header.imports_offset.get(endian) as usize..)
69            .read_error("Invalid Mach-O chained fixups imports offset")?;
70        let symbols = data
71            .get(self.header.symbols_offset.get(endian) as usize..)
72            .read_error("Invalid Mach-O chained fixups symbols offset")?;
73        Ok(DyldChainedImportIterator {
74            endian,
75            imports: Bytes(imports),
76            symbols: Bytes(symbols),
77            count: self.header.imports_count.get(endian),
78            format: self.header.imports_format.get(endian),
79        })
80    }
81
82    /// Return an iterator over the chain starts for each segment.
83    pub fn segments(&self, endian: E) -> Result<DyldChainedSegmentIterator<'data, E>> {
84        let context = "Invalid Mach-O chained starts offset or count";
85        let mut data = self.data;
86        data.skip(self.header.starts_offset.get(endian) as usize)
87            .read_error(context)?;
88
89        let mut header_data = data;
90        let starts_in_image = header_data
91            .read::<macho::DyldChainedStartsInImage<E>>()
92            .read_error(context)?;
93        let seg_count = starts_in_image.seg_count.get(endian);
94        let seg_info_offsets = header_data
95            .read_slice::<U32<E>>(seg_count as usize)
96            .read_error(context)?;
97
98        Ok(DyldChainedSegmentIterator {
99            endian,
100            data,
101            seg_info_offsets,
102            index: 0,
103        })
104    }
105}

119impl<'data, E: Endian> DyldChainedImportIterator<'data, E> {
120    /// Return the next import.
121    pub fn next(&mut self) -> Result<Option<DyldChainedImport<'data>>> {
122        if self.count == 0 {
123            return Ok(None);
124        }
125        let import =
126            DyldChainedImport::parse(self.endian, self.format, self.symbols, &mut self.imports)
127                .map(Some);
128        if import.is_ok() {
129            self.count -= 1;
130        } else {
131            self.count = 0;
132        }
133        import
134    }

145/// An imported symbol in an `LC_DYLD_CHAINED_FIXUPS` load command.
146///
147/// Returned by [`DyldChainedImportIterator::next`].
148#[derive(Debug, Clone, Copy)]
149pub struct DyldChainedImport<'data> {
150    /// The ordinal of the library that the symbol is imported from.
151    pub dylib: macho::BindDylib,
152    /// Whether this is a weak import.
153    pub weak_import: bool,
154    /// The addend to add to the symbol value.
155    pub addend: i64,
156    /// The symbol name.
157    pub name: &'data [u8],
158}
159
160impl<'data> DyldChainedImport<'data> {
161    fn parse<E: Endian>(
162        endian: E,
163        format: macho::DyldChainedImportFormat,
164        symbols: Bytes<'data>,
165        imports: &mut Bytes<'data>,
166    ) -> Result<Self> {
167        let context = "Invalid Mach-O chained fixups import";
168        let (dylib, weak_import, name_offset, addend) = match format {
169            macho::DYLD_CHAINED_IMPORT | macho::DYLD_CHAINED_IMPORT_ADDEND => {
170                let import = imports
171                    .read::<U32<E, macho::DyldChainedImport32>>()
172                    .read_error(context)?
173                    .get(endian);
174                let addend = if format == macho::DYLD_CHAINED_IMPORT_ADDEND {
175                    imports.read::<I32<E>>().read_error(context)?.get(endian)
176                } else {
177                    0
178                };
179                (
180                    import.dylib(),
181                    import.weak_import(),
182                    import.name_offset(),
183                    i64::from(addend),
184                )
185            }
186            macho::DYLD_CHAINED_IMPORT_ADDEND64 => {
187                let import = imports
188                    .read::<U64<E, macho::DyldChainedImport64>>()
189                    .read_error(context)?
190                    .get(endian);
191                let addend = imports.read::<U64<E>>().read_error(context)?.get(endian);
192                (
193                    import.dylib(),
194                    import.weak_import(),
195                    import.name_offset(),
196                    addend as i64,
197                )
198            }
199            _ => return Err(Error("Unsupported Mach-O chained fixups import format")),
200        };
201        let name = symbols
202            .read_string_at(name_offset as usize)
203            .read_error("Invalid Mach-O chained fixups import name offset")?;
204        Ok(DyldChainedImport {
205            dylib,
206            weak_import,
207            addend,
208            name,
209        })
210    }
211}

213/// An iterator over the chain starts for each segment.
214///
215/// Returned by [`DyldChainedFixups::segments`]. Segments without any fixups are skipped.
216#[derive(Debug, Clone, Copy)]
217pub struct DyldChainedSegmentIterator<'data, E: Endian> {
218    endian: E,
219    data: Bytes<'data>,
220    seg_info_offsets: &'data [U32<E>],
221    index: u32,
222}
223
224impl<'data, E: Endian> DyldChainedSegmentIterator<'data, E> {
225    /// Return the chain starts for the next segment that has fixups.
226    pub fn next(&mut self) -> Result<Option<DyldChainedSegment<'data, E>>> {
227        let context = "Invalid Mach-O chained starts in segment";
228        while let Some(seg_info_offset) = self.seg_info_offsets.get(self.index as usize) {
229            let index = self.index;
230            self.index += 1;
231            let seg_info_offset = seg_info_offset.get(self.endian);
232            if seg_info_offset == 0 {
233                // No fixups in this segment.
234                continue;
235            }
236            let mut data = self.data;
237            data.skip(seg_info_offset as usize).read_error(context)?;
238            let header = data
239                .read::<macho::DyldChainedStartsInSegment<E>>()
240                .read_error(context)?;
241            let starts_size = header
242                .size
243                .get(self.endian)
244                .checked_sub(mem::size_of_val(header) as u32)
245                .read_error(context)?;
246            let starts = data
247                .read_slice::<U16<E>>(starts_size as usize / 2)
248                .read_error(context)?;
249            return Ok(Some(DyldChainedSegment {
250                index,
251                header,
252                starts,
253            }));
254        }
255        Ok(None)
256    }
257}

344    page_index: usize,

353impl<'data, E: Endian> DyldChainedFixupIterator<'data, E> {
354    /// Return the next fixup in the segment.
355    ///
356    /// Returns the offset of the fixup within the segment, and the decoded fixup.
357    pub fn next(&mut self) -> Result<Option<(u64, Fixup)>> {
358        let result = self.next_inner();
359        if result.is_err() {
360            self.state = FixupState::Start;
361            self.page_index = self.page_count;
362        }
363        result
364    }
365
366    fn next_inner(&mut self) -> Result<Option<(u64, Fixup)>> {
367        loop {
368            match self.state {
369                FixupState::Start => {
370                    if self.page_index >= self.page_count {
371                        return Ok(None);
372                    }
373                    let page_start = self
374                        .starts
375                        .get(self.page_index)
376                        .read_error("Invalid Mach-O chained fixup page count")?
377                        .get(self.endian);
378                    self.page_offset = self.page_index as u64 * u64::from(self.page_size);
379                    self.page_index += 1;
380                    if page_start == macho::DYLD_CHAINED_PTR_START_NONE {
381                        // No fixups for this page.
382                    } else if page_start & macho::DYLD_CHAINED_PTR_START_MULTI != 0 {
383                        self.state = FixupState::Chain;
384                        self.chain_index =
385                            usize::from(page_start & !macho::DYLD_CHAINED_PTR_START_MULTI);
386                    } else {
387                        self.state = FixupState::Page;
388                        self.offset = self.page_offset + u64::from(page_start);
389                    }
390                }
391                FixupState::Chain => {
392                    let chain_start = self
393                        .starts
394                        .get(self.chain_index)
395                        .read_error("Invalid Mach-O chained fixup chain index")?
396                        .get(self.endian);
397                    self.chain_index += 1;
398                    self.offset = self.page_offset
399                        + u64::from(chain_start & !macho::DYLD_CHAINED_PTR_START_LAST);
400                    if chain_start & macho::DYLD_CHAINED_PTR_START_LAST != 0 {
401                        self.state = FixupState::Page;
402                    } else {
403                        self.state = FixupState::PageChain;
404                    }
405                }
406                FixupState::Page | FixupState::PageChain => {
407                    let offset = self.offset;
408                    let pointer = match self.pointer_format {
409                        macho::DYLD_CHAINED_PTR_32
410                        | macho::DYLD_CHAINED_PTR_32_CACHE
411                        | macho::DYLD_CHAINED_PTR_32_FIRMWARE => u64::from(
412                            self.segment_data
413                                .read_at::<U32<E>>(offset as usize)
414                                .read_error("Invalid Mach-O chained fixup offset")?
415                                .get(self.endian),
416                        ),
417                        _ => self
418                            .segment_data
419                            .read_at::<U64<E>>(offset as usize)
420                            .read_error("Invalid Mach-O chained fixup offset")?
421                            .get(self.endian),
422                    };
423                    let (next, fixup) =
424                        Fixup::parse(self.pointer_format, pointer, self.preferred_load_address)?;
425                    if next == 0 {
426                        if self.state == FixupState::PageChain {
427                            self.state = FixupState::Chain
428                        } else {
429                            self.state = FixupState::Start
430                        };
431                    } else {
432                        self.offset = offset + next;
433                    }
434                    return Ok(Some((offset, fixup)));
435                }
436            }
437        }
438    }
439}

517impl Fixup {
518    fn parse(format: macho::DyldChainedPtrFormat, value: u64, vmaddr: u64) -> Result<(u64, Self)> {
519        match format {
520            macho::DYLD_CHAINED_PTR_ARM64E => Ok(Fixup::parse_arm64e(value, 8, vmaddr)),
521            macho::DYLD_CHAINED_PTR_ARM64E_KERNEL => Ok(Fixup::parse_arm64e(value, 4, 0)),
522            macho::DYLD_CHAINED_PTR_ARM64E_USERLAND => Ok(Fixup::parse_arm64e(value, 8, 0)),
523            macho::DYLD_CHAINED_PTR_ARM64E_FIRMWARE => Ok(Fixup::parse_arm64e(value, 4, vmaddr)),
524            macho::DYLD_CHAINED_PTR_ARM64E_USERLAND24 => Ok(Fixup::parse_userland24(value)),
525            macho::DYLD_CHAINED_PTR_64 => Ok(Fixup::parse_64(value, vmaddr)),
526            macho::DYLD_CHAINED_PTR_64_OFFSET => Ok(Fixup::parse_64(value, 0)),
527            macho::DYLD_CHAINED_PTR_64_KERNEL_CACHE => Ok(Fixup::parse_64_kernel_cache(value, 4)),
528            macho::DYLD_CHAINED_PTR_X86_64_KERNEL_CACHE => {
529                Ok(Fixup::parse_64_kernel_cache(value, 1))
530            }
531            macho::DYLD_CHAINED_PTR_32 => Ok(Fixup::parse_32(value, vmaddr)),
532            macho::DYLD_CHAINED_PTR_32_CACHE => Ok(Fixup::parse_32_cache(value)),
533            macho::DYLD_CHAINED_PTR_32_FIRMWARE => Ok(Fixup::parse_32_firmware(value)),
534            macho::DYLD_CHAINED_PTR_ARM64E_SEGMENTED => Ok(Fixup::parse_arm64e_segmented(value)),
535            macho::DYLD_CHAINED_PTR_ARM64E_SHARED_CACHE => {
536                Ok(Fixup::parse_arm64e_shared_cache(value))
537            }
538            _ => Err(Error("Unsupported Mach-O chained pointer format")),
539        }
540    }
541
542    fn parse_arm64e(ptr: u64, stride: u64, vmaddr: u64) -> (u64, Self) {
543        let ptr = macho::DyldChainedPtrArm64e(ptr);
544        let fixup = if ptr.is_bind() {
545            Self::parse_arm64e_bind(ptr)
546        } else {
547            Self::parse_arm64e_rebase(ptr, vmaddr)
548        };
549        (ptr.next() * stride, fixup)
550    }
551
552    fn parse_arm64e_bind(ptr: macho::DyldChainedPtrArm64e) -> Self {
553        if ptr.is_auth() {
554            let ptr = ptr.auth_bind();
555            let key = match ptr.key() {
556                1 => macho::PtrauthKey::IB,
557                2 => macho::PtrauthKey::DA,
558                3 => macho::PtrauthKey::DB,
559                _ => macho::PtrauthKey::IA,
560            };
561            let auth = Some(FixupAuth {
562                key,
563                diversity: ptr.diversity(),
564                addr_div: ptr.addr_div(),
565            });
566            Fixup::Bind(FixupBind {
567                ordinal: ptr.ordinal(),
568                addend: 0,
569                auth,
570            })
571        } else {
572            let ptr = ptr.bind();
573            Fixup::Bind(FixupBind {
574                ordinal: ptr.ordinal(),
575                addend: ptr.addend(),
576                auth: None,
577            })
578        }
579    }
580
581    fn parse_arm64e_rebase(ptr: macho::DyldChainedPtrArm64e, vmaddr: u64) -> Self {
582        if ptr.is_auth() {
583            let ptr = ptr.auth_rebase();
584            let target_offset = ptr.runtime_offset();
585            let key = match ptr.key() {
586                1 => macho::PtrauthKey::IB,
587                2 => macho::PtrauthKey::DA,
588                3 => macho::PtrauthKey::DB,
589                _ => macho::PtrauthKey::IA,
590            };
591            let auth = Some(FixupAuth {
592                key,
593                diversity: ptr.diversity(),
594                addr_div: ptr.addr_div(),
595            });
596            Fixup::Rebase(FixupRebase {
597                target_offset,
598                auth,
599            })
600        } else {
601            let ptr = ptr.rebase();
602            let target_offset = ptr.target().wrapping_sub(vmaddr) | ptr.high8() << 56;
603            Fixup::Rebase(FixupRebase {
604                target_offset,
605                auth: None,
606            })
607        }
608    }
609
610    fn parse_userland24(ptr: u64) -> (u64, Self) {
611        let ptr = macho::DyldChainedPtrArm64e(ptr);
612        let fixup = if ptr.is_bind() {
613            Self::parse_arm64e_bind24(ptr)
614        } else {
615            Self::parse_arm64e_rebase(ptr, 0)
616        };
617        (ptr.next() * 8, fixup)
618    }
619
620    fn parse_arm64e_bind24(ptr: macho::DyldChainedPtrArm64e) -> Self {
621        if ptr.is_auth() {
622            let ptr = ptr.auth_bind24();
623            let key = match ptr.key() {
624                1 => macho::PtrauthKey::IB,
625                2 => macho::PtrauthKey::DA,

628            };
629            let auth = Some(FixupAuth {

652            let ptr = ptr.bind();
653            Fixup::Bind(FixupBind {
654                ordinal: ptr.ordinal(),
655                addend: ptr.addend(),
656                auth: None,
657            })
658        } else {
659            let ptr = ptr.rebase();
660            let target_offset = ptr.target().wrapping_sub(vmaddr) | ptr.high8() << 56;
661            Fixup::Rebase(FixupRebase {
662                target_offset,
663                auth: None,
664            })
665        };
666        (ptr.next() * 4, fixup)
667    }
