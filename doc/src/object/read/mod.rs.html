67/// The error type used within the read module.
68#[derive(Debug, Clone, Copy, PartialEq, Eq)]
69pub struct Error(&'static str);
70
71impl fmt::Display for Error {
72    #[inline]
73    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
74        f.write_str(self.0)
75    }
76}
77
78#[cfg(feature = "std")]
79impl std::error::Error for Error {}

81/// The result type used within the read module.
82pub type Result<T> = result::Result<T, Error>;
83
84trait ReadError<T> {
85    fn read_error(self, error: &'static str) -> Result<T>;
86}
87
88impl<T> ReadError<T> for result::Result<T, ()> {
89    fn read_error(self, error: &'static str) -> Result<T> {
90        self.map_err(|()| Error(error))
91    }
92}
93
94impl<T> ReadError<T> for result::Result<T, Error> {
95    fn read_error(self, error: &'static str) -> Result<T> {
96        self.map_err(|_| Error(error))
97    }
98}
99
100impl<T> ReadError<T> for Option<T> {
101    fn read_error(self, error: &'static str) -> Result<T> {
102        self.ok_or(Error(error))
103    }

200impl FileKind {
201    /// Determine a file kind by parsing the start of the file.
202    pub fn parse<'data, R: ReadRef<'data>>(data: R) -> Result<FileKind> {
203        Self::parse_at(data, 0)
204    }
205
206    /// Determine a file kind by parsing at the given offset.
207    pub fn parse_at<'data, R: ReadRef<'data>>(data: R, offset: u64) -> Result<FileKind> {
208        let magic = data
209            .read_bytes_at(offset, 16)
210            .read_error("Could not read file magic")?;
211        if magic.len() < 16 {
212            return Err(Error("File too short"));
213        }
214
215        let kind = match [magic[0], magic[1], magic[2], magic[3], magic[4], magic[5], magic[6], magic[7]] {
216            #[cfg(feature = "archive")]
217            [b'!', b'<', b'a', b'r', b'c', b'h', b'>', b'\n'] => FileKind::Archive,
218            #[cfg(feature = "macho")]
219            [b'd', b'y', b'l', b'd', b'_', b'v', b'1', b' '] => FileKind::DyldCache,
220            #[cfg(feature = "elf")]
221            [0x7f, b'E', b'L', b'F', 1, ..] => FileKind::Elf32,
222            #[cfg(feature = "elf")]
223            [0x7f, b'E', b'L', b'F', 2, ..] => FileKind::Elf64,
224            #[cfg(feature = "macho")]
225            [0xfe, 0xed, 0xfa, 0xce, ..]
226            | [0xce, 0xfa, 0xed, 0xfe, ..] => FileKind::MachO32,
227            #[cfg(feature = "macho")]
228            | [0xfe, 0xed, 0xfa, 0xcf, ..]
229            | [0xcf, 0xfa, 0xed, 0xfe, ..] => FileKind::MachO64,
230            #[cfg(feature = "macho")]
231            [0xca, 0xfe, 0xba, 0xbe, ..] => FileKind::MachOFat32,
232            #[cfg(feature = "macho")]
233            [0xca, 0xfe, 0xba, 0xbf, ..] => FileKind::MachOFat64,
234            #[cfg(feature = "wasm")]
235            [0x00, b'a', b's', b'm', ..] => FileKind::Wasm,
236            #[cfg(feature = "pe")]
237            [b'M', b'Z', ..] if offset == 0 => {
238                // offset == 0 restriction is because optional_header_magic only looks at offset 0
239                match pe::optional_header_magic(data) {
240                    Ok(crate::pe::IMAGE_NT_OPTIONAL_HDR32_MAGIC) => {
241                        FileKind::Pe32
242                    }
243                    Ok(crate::pe::IMAGE_NT_OPTIONAL_HDR64_MAGIC) => {
244                        FileKind::Pe64
245                    }
246                    _ => return Err(Error("Unknown MS-DOS file")),
247                }
248            }
249            // TODO: more COFF machines
250            #[cfg(feature = "coff")]
251            // COFF arm
252            [0xc4, 0x01, ..]
253            // COFF arm64
254            | [0x64, 0xaa, ..]
255            // COFF x86
256            | [0x4c, 0x01, ..]
257            // COFF x86-64
258            | [0x64, 0x86, ..] => FileKind::Coff,
259            #[cfg(feature = "coff")]
260            [0x00, 0x00, 0xff, 0xff, 0x00, 0x00, ..] => FileKind::CoffImport,
261            #[cfg(feature = "coff")]
262            [0x00, 0x00, 0xff, 0xff, 0x02, 0x00, ..] if offset == 0 => {
263                // offset == 0 restriction is because anon_object_class_id only looks at offset 0
264                match coff::anon_object_class_id(data) {
265                    Ok(crate::pe::ANON_OBJECT_HEADER_BIGOBJ_CLASS_ID) => FileKind::CoffBig,
266                    _ => return Err(Error("Unknown anon object file")),
267                }
268            }
269            #[cfg(feature = "xcoff")]
270            [0x01, 0xdf, ..] => FileKind::Xcoff32,
271            #[cfg(feature = "xcoff")]
272            [0x01, 0xf7, ..] => FileKind::Xcoff64,
273            _ => return Err(Error("Unknown file magic")),
274        };
275        Ok(kind)
276    }
277}

341/// A map from addresses to symbols.
342#[derive(Debug, Default, Clone)]
343pub struct SymbolMap<T: SymbolMapEntry> {
344    symbols: Vec<T>,
345}
346
347impl<T: SymbolMapEntry> SymbolMap<T> {
348    /// Construct a new symbol map.
349    ///
350    /// This function will sort the symbols by address.
351    pub fn new(mut symbols: Vec<T>) -> Self {
352        symbols.sort_unstable_by_key(|s| s.address());
353        SymbolMap { symbols }
354    }
355
356    /// Get the symbol before the given address.
357    pub fn get(&self, address: u64) -> Option<&T> {
358        let index = match self
359            .symbols
360            .binary_search_by_key(&address, |symbol| symbol.address())
361        {
362            Ok(index) => index,
363            Err(index) => index.checked_sub(1)?,
364        };
365        self.symbols.get(index)
366    }
367
368    /// Get all symbols in the map.
369    #[inline]
370    pub fn symbols(&self) -> &[T] {
371        &self.symbols
372    }
373}

648/// A range in a file that may be compressed.
649#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
650pub struct CompressedFileRange {
651    /// The data compression format.
652    pub format: CompressionFormat,
653    /// The file offset of the compressed data.
654    pub offset: u64,
655    /// The compressed data size.
656    pub compressed_size: u64,
657    /// The uncompressed data size.
658    pub uncompressed_size: u64,
659}
660
661impl CompressedFileRange {
662    /// Data that is uncompressed.
663    #[inline]
664    pub fn none(range: Option<(u64, u64)>) -> Self {
665        if let Some((offset, size)) = range {
666            CompressedFileRange {
667                format: CompressionFormat::None,
668                offset,
669                compressed_size: size,
670                uncompressed_size: size,
671            }
672        } else {
673            CompressedFileRange {
674                format: CompressionFormat::None,
675                offset: 0,
676                compressed_size: 0,
677                uncompressed_size: 0,
678            }
679        }
680    }
681
682    /// Convert to `CompressedData` by reading from the file.
683    pub fn data<'data, R: ReadRef<'data>>(self, file: R) -> Result<CompressedData<'data>> {
684        let data = file
685            .read_bytes_at(self.offset, self.compressed_size)
686            .read_error("Invalid compressed data size or offset")?;
687        Ok(CompressedData {
688            format: self.format,
689            data,
690            uncompressed_size: self.uncompressed_size,
691        })
692    }
693}

695/// Data that may be compressed.
696#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
697pub struct CompressedData<'data> {
698    /// The data compression format.
699    pub format: CompressionFormat,
700    /// The compressed data.
701    pub data: &'data [u8],
702    /// The uncompressed data size.
703    pub uncompressed_size: u64,
704}
705
706impl<'data> CompressedData<'data> {
707    /// Data that is uncompressed.
708    #[inline]
709    pub fn none(data: &'data [u8]) -> Self {
710        CompressedData {
711            format: CompressionFormat::None,
712            data,
713            uncompressed_size: data.len() as u64,
714        }
715    }
716
717    /// Return the uncompressed data.
718    ///
719    /// Returns an error for invalid data or unsupported compression.
720    /// This includes if the data is compressed but the `compression` feature
721    /// for this crate is disabled.
722    pub fn decompress(self) -> Result<Cow<'data, [u8]>> {
723        match self.format {
724            CompressionFormat::None => Ok(Cow::Borrowed(self.data)),
725            #[cfg(feature = "compression")]
726            CompressionFormat::Zlib => {
727                use core::convert::TryInto;
728                let size = self
729                    .uncompressed_size
730                    .try_into()
731                    .ok()
732                    .read_error("Uncompressed data size is too large.")?;
733                let mut decompressed = Vec::with_capacity(size);
734                let mut decompress = flate2::Decompress::new(true);
735                decompress
736                    .decompress_vec(
737                        self.data,
738                        &mut decompressed,
739                        flate2::FlushDecompress::Finish,
740                    )
741                    .ok()
742                    .read_error("Invalid zlib compressed data")?;
743                Ok(Cow::Owned(decompressed))
744            }
745            #[cfg(feature = "compression")]
746            CompressionFormat::Zstandard => {
747                use core::convert::TryInto;
748                use std::io::Read;
749                let size = self
750                    .uncompressed_size
751                    .try_into()
752                    .ok()
753                    .read_error("Uncompressed data size is too large.")?;
754                let mut decompressed = Vec::with_capacity(size);
755                let mut decoder = ruzstd::StreamingDecoder::new(self.data)
756                    .ok()
757                    .read_error("Invalid zstd compressed data")?;
758                decoder
759                    .read_to_end(&mut decompressed)
760                    .ok()
761                    .read_error("Invalid zstd compressed data")?;
762                Ok(Cow::Owned(decompressed))
763            }
764            _ => Err(Error("Unsupported compressed data.")),
765        }
766    }
767}
