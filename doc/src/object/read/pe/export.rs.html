48/// Where an export is pointing to.
49#[derive(Clone, Copy)]
50pub enum ExportTarget<'data> {
51    /// The address of the export, relative to the image base.
52    Address(u32),
53    /// Forwarded to an export ordinal in another DLL.
54    ///
55    /// This gives the name of the DLL, and the ordinal.
56    ForwardByOrdinal(&'data [u8], ExportOrdinal),
57    /// Forwarded to an export name in another DLL.
58    ///
59    /// This gives the name of the DLL, and the export name.
60    ForwardByName(&'data [u8], &'data [u8]),
61}
62
63impl<'data> ExportTarget<'data> {
64    /// Returns true if the target is an address.
65    pub fn is_address(&self) -> bool {
66        match self {
67            ExportTarget::Address(_) => true,
68            _ => false,
69        }
70    }
71
72    /// Returns true if the export is forwarded to another DLL.
73    pub fn is_forward(&self) -> bool {
74        !self.is_address()
75    }
76}

78/// An export from a PE file.
79///
80/// There are multiple kinds of PE exports (with or without a name, and local or forwarded).
81#[derive(Clone, Copy)]
82pub struct Export<'data> {
83    /// The ordinal of the export.
84    ///
85    /// These are sequential, starting at a base specified in the DLL.
86    pub ordinal: ExportOrdinal,
87    /// The name of the export, if known.
88    pub name: Option<&'data [u8]>,
89    /// The target of this export.
90    pub target: ExportTarget<'data>,
91}

98            .field("target", &self.target)
99            .finish()
100    }

103impl<'a> fmt::Debug for ExportTarget<'a> {
104    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::result::Result<(), core::fmt::Error> {
105        match self {
106            ExportTarget::Address(address) => write!(f, "Address({:#x})", address),
107            ExportTarget::ForwardByOrdinal(library, ordinal) => write!(
108                f,
109                "ForwardByOrdinal({:?}.#{})",
110                ByteString(library),
111                ordinal
112            ),
113            ExportTarget::ForwardByName(library, name) => write!(
114                f,
115                "ForwardByName({:?}.{:?})",
116                ByteString(library),
117                ByteString(name)
118            ),
119        }
120    }

123/// A partially parsed PE export table.
124///
125/// Returned by [`DataDirectories::export_table`](super::DataDirectories::export_table).
126#[derive(Debug, Clone)]
127pub struct ExportTable<'data> {
128    data: Bytes<'data>,
129    virtual_address: u32,
130    directory: &'data pe::ImageExportDirectory,
131    ordinal_base: u16,
132    addresses: &'data [U32<LE>],
133    names: &'data [U32<LE>],
134    name_ordinals: &'data [U16<LE, ExportAddressIndex>],
135}
136
137impl<'data> ExportTable<'data> {
138    /// Parse the export table given its section data and address.
139    pub fn parse(data: &'data [u8], virtual_address: u32) -> Result<Self> {
140        let directory = Self::parse_directory(data)?;
141        let data = Bytes(data);
142
143        let Ok(ordinal_base) = u16::try_from(directory.base.get(LE)) else {
144            return Err(Error("Invalid PE export ordinal base"));
145        };
146
147        let mut addresses = &[][..];
148        let address_of_functions = directory.address_of_functions.get(LE);
149        if address_of_functions != 0 {
150            let number = directory.number_of_functions.get(LE) as usize;
151            // Ordinals must fit in a u16, so this bounds every valid address index too.
152            let max_index = usize::from(u16::MAX - ordinal_base);
153            if number > max_index + 1 {
154                return Err(Error("Invalid PE export number of functions"));
155            }
156            addresses = data
157                .read_slice_at::<U32<_>>(
158                    address_of_functions.wrapping_sub(virtual_address) as usize,
159                    number,
160                )
161                .read_error("Invalid PE export address table")?;
162        }
163
164        let mut names = &[][..];
165        let mut name_ordinals = &[][..];
166        let address_of_names = directory.address_of_names.get(LE);
167        let address_of_name_ordinals = directory.address_of_name_ordinals.get(LE);
168        if address_of_names != 0 {
169            if address_of_name_ordinals == 0 {
170                return Err(Error("Missing PE export ordinal table"));
171            }
172
173            let number = directory.number_of_names.get(LE) as usize;
174            names = data
175                .read_slice_at::<U32<_>>(
176                    address_of_names.wrapping_sub(virtual_address) as usize,
177                    number,
178                )
179                .read_error("Invalid PE export name pointer table")?;
180            name_ordinals = data
181                .read_slice_at::<U16<_, _>>(
182                    address_of_name_ordinals.wrapping_sub(virtual_address) as usize,
183                    number,
184                )
185                .read_error("Invalid PE export ordinal table")?;
186        }
187
188        Ok(ExportTable {
189            data,
190            virtual_address,
191            directory,
192            ordinal_base,
193            addresses,
194            names,
195            name_ordinals,
196        })
197    }

199    /// Parse the export directory given its section data.
200    pub fn parse_directory(data: &'data [u8]) -> Result<&'data pe::ImageExportDirectory> {
201        data.read_at::<pe::ImageExportDirectory>(0)
202            .read_error("Invalid PE export dir size")
203    }

205    /// Returns the header of the export table.
206    pub fn directory(&self) -> &'data pe::ImageExportDirectory {
207        self.directory
208    }

210    /// Returns the base value of ordinals.
211    ///
212    /// Adding this to an address index will give an ordinal.
213    pub fn ordinal_base(&self) -> u16 {
214        self.ordinal_base
215    }
216
217    /// Add the ordinal base to an address index.
218    ///
219    /// Returns an error if the index is out of bounds for the address table.
220    pub fn ordinal_from_index(&self, index: ExportAddressIndex) -> Result<ExportOrdinal> {
221        if usize::from(index.0) >= self.addresses.len() {
222            return Err(Error("Invalid PE export address index"));
223        }
224        Ok(ExportOrdinal(self.ordinal_base + index.0))
225    }
226
227    /// Subtract the ordinal base from an ordinal to obtain an address index.
228    ///
229    /// Returns an error if the resulting index is out of bounds for the address table.
230    pub fn index_from_ordinal(&self, ordinal: ExportOrdinal) -> Result<ExportAddressIndex> {
231        let index = ordinal.0.wrapping_sub(self.ordinal_base);
232        if usize::from(index) >= self.addresses.len() {
233            return Err(Error("Invalid PE export ordinal"));
234        }
235        Ok(ExportAddressIndex(index))
236    }

238    /// Returns the unparsed address table.
239    ///
240    /// An address table entry may be a local address, or the address of a forwarded export entry.
241    /// See [`Self::is_forward`] and [`Self::target_from_address`].
242    pub fn addresses(&self) -> &'data [U32<LE>] {
243        self.addresses
244    }

246    /// Returns an iterator for the entries in the address table.
247    pub fn address_iter(
248        &self,
249    ) -> impl Iterator<Item = (ExportAddressIndex, ExportOrdinal, u32)> + use<'data> {
250        let ordinal_base = self.ordinal_base;
251        self.addresses.iter().enumerate().map(move |(i, x)| {
252            (
253                ExportAddressIndex(i as u16),
254                ExportOrdinal(ordinal_base + i as u16),
255                x.get(LE),
256            )
257        })
258    }

260    /// Returns the unparsed name pointer table.
261    ///
262    /// A name pointer table entry can be used with [`Self::name_from_pointer`].
263    pub fn name_pointers(&self) -> &'data [U32<LE>] {
264        self.names
265    }
266
267    /// Returns the unparsed ordinal table.
268    ///
269    /// An ordinal table entry is a 0-based index into the address table.
270    /// See [`Self::address_by_index`] and [`Self::target_by_index`].
271    pub fn name_ordinals(&self) -> &'data [U16<LE, ExportAddressIndex>] {
272        self.name_ordinals
273    }
274
275    /// Returns an iterator for the entries in the name pointer table and ordinal table.
276    ///
277    /// A name pointer table entry can be used with [`Self::name_from_pointer`].
278    ///
279    /// An ordinal table entry is a 0-based index into the address table.
280    /// See [`Self::address_by_index`] and [`Self::target_by_index`].
281    pub fn name_iter(&self) -> impl Iterator<Item = (u32, ExportAddressIndex)> + use<'data> {
282        self.names
283            .iter()
284            .map(|x| x.get(LE))
285            .zip(self.name_ordinals.iter().map(|x| x.get(LE)))
286    }

288    /// Returns the export address table entry at the given address index.
289    ///
290    /// This may be a local address, or the address of a forwarded export entry.
291    /// See [`Self::is_forward`] and [`Self::target_from_address`].
292    ///
293    /// `index` is a 0-based index into the export address table.
294    pub fn address_by_index(&self, index: ExportAddressIndex) -> Result<u32> {
295        Ok(self
296            .addresses
297            .get(index.0 as usize)
298            .read_error("Invalid PE export address index")?
299            .get(LE))
300    }

302    /// Returns the export address table entry at the given ordinal.
303    ///
304    /// This may be a local address, or the address of a forwarded export entry.
305    /// See [`Self::is_forward`] and [`Self::target_from_address`].
306    pub fn address_by_ordinal(&self, ordinal: ExportOrdinal) -> Result<u32> {
307        let index = self.index_from_ordinal(ordinal)?;
308        self.address_by_index(index)
309    }

311    /// Returns the target of the export at the given address index.
312    ///
313    /// `index` is a 0-based index into the export address table.
314    pub fn target_by_index(&self, index: ExportAddressIndex) -> Result<ExportTarget<'data>> {
315        self.target_from_address(self.address_by_index(index)?)
316    }
317
318    /// Returns the target of the export at the given ordinal.
319    pub fn target_by_ordinal(&self, ordinal: ExportOrdinal) -> Result<ExportTarget<'data>> {
320        self.target_from_address(self.address_by_ordinal(ordinal)?)
321    }

323    /// Convert an export address table entry into a target.
324    pub fn target_from_address(&self, address: u32) -> Result<ExportTarget<'data>> {
325        Ok(if let Some(forward) = self.forward_string(address)? {
326            let i = forward
327                .iter()
328                .rposition(|x| *x == b'.')
329                .read_error("Missing PE forwarded export separator")?;
330            let library = &forward[..i];
331            match &forward[i + 1..] {
332                [b'#', digits @ ..] => {
333                    let ordinal =
334                        parse_ordinal(digits).read_error("Invalid PE forwarded export ordinal")?;
335                    ExportTarget::ForwardByOrdinal(library, ordinal)
336                }
337                [] => {
338                    return Err(Error("Missing PE forwarded export name"));
339                }
340                name => ExportTarget::ForwardByName(library, name),
341            }
342        } else {
343            ExportTarget::Address(address)
344        })
345    }
346
347    fn forward_offset(&self, address: u32) -> Option<usize> {
348        let offset = address.wrapping_sub(self.virtual_address) as usize;
349        if offset < self.data.len() {
350            Some(offset)
351        } else {
352            None
353        }
354    }
355
356    /// Return true if the export address table entry is a forward.
357    pub fn is_forward(&self, address: u32) -> bool {
358        self.forward_offset(address).is_some()
359    }
360
361    /// Return the forward string if the export address table entry is a forward.
362    pub fn forward_string(&self, address: u32) -> Result<Option<&'data [u8]>> {
363        if let Some(offset) = self.forward_offset(address) {
364            self.data
365                .read_string_at(offset)
366                .read_error("Invalid PE forwarded export address")
367                .map(Some)
368        } else {
369            Ok(None)
370        }
371    }
372
373    /// Convert an export name pointer table entry into a name.
374    pub fn name_from_pointer(&self, name_pointer: u32) -> Result<&'data [u8]> {
375        let offset = name_pointer.wrapping_sub(self.virtual_address);
376        self.data
377            .read_string_at(offset as usize)
378            .read_error("Invalid PE export name pointer")
379    }
380
381    /// Returns the parsed exports in this table.
382    pub fn exports(&self) -> Result<Vec<Export<'data>>> {
383        // First, let's list all exports.
384        let mut exports = Vec::new();
385        for (_index, ordinal, address) in self.address_iter() {
386            let target = self.target_from_address(address)?;
387            exports.push(Export {
388                ordinal,
389                target,
390                // Might be populated later.
391                name: None,
392            });
393        }
394
395        // Now, check whether some (or all) of them have an associated name.
396        // `ordinal_index` is a 0-based index into `addresses`.
397        for (name_pointer, ordinal_index) in self.name_iter() {
398            let name = self.name_from_pointer(name_pointer)?;
399            exports
400                .get_mut(ordinal_index.0 as usize)
401                .read_error("Invalid PE export ordinal")?
402                .name = Some(name);
403        }
404
405        Ok(exports)
406    }
407}

409fn parse_ordinal(digits: &[u8]) -> Option<ExportOrdinal> {
410    if digits.is_empty() {
411        return None;
412    }
413    let mut result: u16 = 0;
414    for &c in digits {
415        let x = (c as char).to_digit(10)? as u16;
416        result = result.checked_mul(10)?.checked_add(x)?;
417    }
418    Some(ExportOrdinal(result))
419}

421/// An iterator for the exports in a [`PeFile`].
422pub struct PeExportIterator<'data, 'file, R = &'data [u8]>
423where
424    R: ReadRef<'data>,
425{
426    image_base: u64,
427    table: Option<ExportTable<'data>>,
428    index: usize,
429    // false: iterating names, true: iterating addresses
430    addresses: bool,
431    seen: Vec<bool>,
432    marker: PhantomData<(&'file (), R)>,
433}
434
435impl<'data, 'file, R> PeExportIterator<'data, 'file, R>
436where
437    R: ReadRef<'data>,
438{
439    pub(super) fn new<Pe: ImageNtHeaders>(file: &'file PeFile<'data, Pe, R>) -> Result<Self> {
440        let table = file.export_table()?;
441        let seen = match &table {
442            Some(table) => vec![false; table.addresses().len()],
443            None => Vec::new(),
444        };
445        Ok(PeExportIterator {
446            image_base: file.common.image_base,
447            table,
448            index: 0,
449            addresses: false,
450            seen,
451            marker: PhantomData,
452        })
453    }
454
455    fn next(&mut self) -> read::Result<Option<read::Export<'data>>> {
456        let Some(table) = &self.table else {
457            return Ok(None);
458        };
459        if !self.addresses {
460            let index = self.index;
461            if let (Some(name_pointer), Some(address_index)) = (
462                table.name_pointers().get(index),
463                table.name_ordinals().get(index),
464            ) {
465                // Ensure progress is made, so errors after here don't need to terminate iteration.
466                self.index += 1;
467
468                let address_index = address_index.get(LE);
469                if let Some(seen) = self.seen.get_mut(usize::from(address_index.0)) {
470                    *seen = true;
471                }
472
473                let address = table.address_by_index(address_index)?;
474                let ordinal = table.ordinal_from_index(address_index)?;
475                let name = table.name_from_pointer(name_pointer.get(LE))?;
476                let name = NameOrOrdinal::Name(Cow::Borrowed(name));
477                return self.export(table, address, ordinal, name);
478            } else {
479                self.index = 0;
480                self.addresses = true;
481            }
482        }
483        loop {
484            let index = self.index;
485            let Some(address) = table.addresses().get(index).map(|x| x.get(LE)) else {
486                return Ok(None);
487            };
488            // Ensure progress is made, so errors after here don't need to terminate iteration.
489            self.index += 1;
490
491            if self.seen[index] || address == 0 {
492                continue;
493            }
494
495            let ordinal = table.ordinal_from_index(ExportAddressIndex(index as u16))?;
496            return self.export(table, address, ordinal, NameOrOrdinal::Ordinal(ordinal.0));
497        }
498    }
499
500    fn export(
501        &self,
502        table: &ExportTable<'data>,
503        address: u32,
504        ordinal: ExportOrdinal,
505        name: NameOrOrdinal<Cow<'data, [u8]>>,
506    ) -> read::Result<Option<read::Export<'data>>> {
507        let target = match table.target_from_address(address)? {
508            ExportTarget::Address(address) => read::ExportTarget::Address {
509                address: self.image_base.wrapping_add(address.into()),
510            },
511            ExportTarget::ForwardByName(library, name) => read::ExportTarget::Reexport {
512                library,
513                name: NameOrOrdinal::Name(name),
514            },
515            ExportTarget::ForwardByOrdinal(library, ordinal) => read::ExportTarget::Reexport {
516                library,
517                name: NameOrOrdinal::Ordinal(ordinal.0),
518            },
519        };
520        Ok(Some(read::Export {
521            name,
522            target,
523            weak: false,
524            flags: read::ExportFlags::Pe { ordinal },
525        }))
526    }
527}
