1//!
2//! This module defines a `BincodeFile` and implements `FileTypeIO<T>` and
3//! `LazyFileTypeIO<T>` trait for a bincode file. It is essentially a wrapper
4//! around the bincode crate. Note that we store additional type info in the file
5//! header. Hence the files cannot be directly parsed using the bincode crate
6//! functions.
7//!
8//! ## Compatibility
9//! Lazily writing items of type T produces a bincode file which **cannot** be read
10//! as `Vec<T>` using `FileTypeIO::read()`, instead you need to use the lazy reader
11//! to read items one at a time. The helper fn `LazyFileTypeIO::read_all()`
12//! can be used to collect all the items in a `Vec<T>`
13//!
14//! ## Simple read/write example
15//! `BincodeFile<T>` implements `FileTypeIO<T>` for any serializable type `T`.
16//! ```rust
17//! use martian_filetypes::{FileTypeRead, FileTypeWrite, bin_file::BincodeFile};
18//! use martian::Error;
19//! use serde::{Serialize, Deserialize};
20//!
21//! #[derive(Debug, PartialEq, Serialize, Deserialize)]
22//! struct Chemistry {
23//!     name: String,
24//!     paired_end: bool,
25//! }
26//!
27//! fn main() -> Result<(), Error> {
28//!     let chem = Chemistry { name: "SCVDJ".into(), paired_end: true };
29//!     let bin_file = BincodeFile::from("bin_example");
30//!     // The two functions below are simple wrappers over bincode crate functions
31//!     bin_file.write(&chem)?;
32//!     let decoded = bin_file.read()?;
33//!     assert_eq!(chem, decoded);
34//!     # std::fs::remove_file(bin_file)?; // Remove the file (hidden from the doc)
35//!     Ok(())
36//! }
37//! ```
38//!
39//! ## Lazy read/write example
40//! If the bincode file stores a list of items of type `T`, then the items can be read
41//! one at a time without reading the whole file into memory. A list of items
42//! of type `T` can also be incrementally written to a bincode file.
43//! `BincodeFile<Vec<T>>` implements `LazyFileTypeIO<T>` for any serializable type `T`.
44//!
45//! ```rust
46//! use martian_filetypes::{FileTypeRead, FileTypeWrite, LazyFileTypeIO, LazyWrite};
47//! use martian_filetypes::bin_file::{BincodeFile, LazyBincodeReader, LazyBincodeWriter};
48//! use martian::Error;
49//! use serde::{Serialize, Deserialize};
50//!
51//! fn main() -> Result<(), Error> {
52//!     let bin_file = BincodeFile::from("bin_example_lazy");
53//!     let mut writer = bin_file.lazy_writer()?;
54//!     // writer implements the trait `LazyWrite<i32>`
55//!     for val in 0..10_000i32 {
56//!         writer.write_item(&val)?;
57//!     }
58//!     writer.finish()?; // The file writing is not completed until finish() is called.
59//!
60//!     // We could have collected the vector and invoked write().
61//!     // Both approaches will give you a bincode file which you can lazily read.
62//!     // Note that the bincode files **will not be identical** in the two cases.
63//!     // let vals: Vec<_> = (0..10_000).into_iter().collect()
64//!     // bin_file.write(&vals)?;
65//!
66//!     let mut reader = bin_file.lazy_reader()?;
67//!     let mut max_val = 0;
68//!     // reader is an `Iterator` over values of type Result<`i32`, Error>
69//!     for (i, val) in reader.enumerate() {
70//!         let val: i32 = val?;
71//!         assert_eq!(i as i32, val);
72//!         max_val = std::cmp::max(max_val, val);
73//!     }
74//!     assert_eq!(max_val, 9999i32);
75//!     # std::fs::remove_file(bin_file)?; // Remove the file (hidden from the doc)
76//!     Ok(())
77//! }
78//! ```

104/// Any type `T` that can be deserialized implements `load()` from a `BincodeFile`
105/// TODO: Include the TypeId here?
106impl<T, F> FileTypeRead<T> for BinaryFormat<F, T>
107where
108    T: Any + DeserializeOwned,
109    F: MartianFileType,
110{
111    fn read_from<R: Read>(mut reader: R) -> Result<T, Error> {
112        fn check_type(
113            actual_type: String,
114            expect_type: &'static str,
115            marker_type: &'static str,
116        ) -> Result<(), Error> {
117            if actual_type == expect_type {
118                Ok(())
119            } else if actual_type == marker_type {
120                Err(format_err!(
121                    "Lazily written bincode files cannot be read using FileTypeIO::read(). Use LazyFileTypeIO::read_all() instead."
122                ))
123            } else {
124                Err(format_err!(
125                    "Data type {} does not match expected type {} or {}",
126                    actual_type,
127                    expect_type,
128                    marker_type
129                ))
130            }
131        }
132        check_type(
133            bincode::deserialize_from(&mut reader)?,
134            type_name::<T>(),
135            type_name::<LazyMarker<T>>(),
136        )?;
137        Ok(bincode::deserialize_from(&mut reader)?)
138    }
139}

161/// Iterator over individual items  within a bincode file that
162/// stores a list of items.
163pub struct LazyBincodeReader<T, F = Bincode, R = BufReader<File>>
164where
165    F: MartianFileType,
166    R: Read,
167    T: Any + DeserializeOwned,
168{
169    reader: R,
170    mode: FileMode,
171    processed_items: usize,
172    phantom_f: PhantomData<F>,
173    phantom_t: PhantomData<T>,
174}
175
176impl<T, F, R> LazyRead<T, R> for LazyBincodeReader<T, F, R>
177where
178    F: MartianFileType,
179    R: Read,
180    T: Any + Serialize + DeserializeOwned,
181{
182    type FileType = BinaryFormat<F, Vec<T>>;
183    fn with_reader(mut reader: R) -> Result<Self, Error> {
184        fn check_type(
185            mut reader: impl Read,
186            vec_type: &'static str,
187            lazy_type: &'static str,
188        ) -> Result<FileMode, Error> {
189            let actual_type: String = bincode::deserialize_from(&mut reader)?;
190            if actual_type == vec_type {
191                let total_items: usize = bincode::deserialize_from(&mut reader)?;
192                Ok(FileMode::Vec(total_items))
193            } else if actual_type == lazy_type {
194                Ok(FileMode::Lazy)
195            } else {
196                Err(format_err!(
197                    "Data type {} does not match expected type {} or {}",
198                    actual_type,
199                    vec_type,
200                    lazy_type
201                ))
202            }
203        }
204        let mode = check_type(
205            &mut reader,
206            type_name::<Vec<T>>(),
207            type_name::<LazyMarker<Vec<T>>>(),
208        )?;
209
210        Ok(LazyBincodeReader {
211            reader,
212            mode,
213            processed_items: 0,
214            phantom_f: PhantomData,
215            phantom_t: PhantomData,
216        })
217    }
218}

220impl<T, F, R> Iterator for LazyBincodeReader<T, F, R>
221where
222    F: MartianFileType,
223    R: Read,
224    T: Any + DeserializeOwned,
225{
226    type Item = Result<T, Error>;
227    fn next(&mut self) -> Option<Self::Item> {
228        match self.mode {
229            FileMode::Vec(total_items) => {
230                if self.processed_items < total_items {
231                    self.processed_items += 1;
232                    match bincode::deserialize_from(&mut self.reader) {
233                        Ok(t) => Some(Ok(t)),
234                        Err(e) => Some(Err(Error::from(e))),
235                    }
236                } else {
237                    None
238                }
239            }
240            FileMode::Lazy => match try_bincode_deserialize(&mut self.reader) {
241                DeserializeResult::Item(t) => Some(Ok(t)),
242                DeserializeResult::Eof => None,
243                DeserializeResult::ReadError(e) => Some(Err(e)),
244            },
245        }
246    }
247}
248
249enum DeserializeResult<T, E> {
250    Item(T),
251    ReadError(E),
252    Eof,
253}
254
255fn try_bincode_deserialize<T, R>(reader: &mut R) -> DeserializeResult<T, Error>
256where
257    R: Read,
258    T: DeserializeOwned,
259{
260    match bincode::deserialize_from(reader) {
261        Ok(t) => DeserializeResult::Item(t),
262        Err(e) => {
263            match *e.as_ref() {
264                bincode::ErrorKind::Io(ref io_e) => {
265                    match io_e.kind() {
266                        io::ErrorKind::UnexpectedEof => DeserializeResult::Eof, // We are at the end of the stream
267                        _ => DeserializeResult::ReadError(Error::from(e)),
268                    }
269                }
270                _ => DeserializeResult::ReadError(Error::from(e)),
271            }
272        }
273    }
274}

276// When a list of items of type T is lazily written,
277// the typeid we store is that of LazyMarker<Vec<T>>
278struct LazyMarker<T>(PhantomData<T>);

280/// Helper struct to write items one by one to a bincode file that
281/// stores a list of items
282pub struct LazyBincodeWriter<T, F = Bincode, W = BufWriter<File>>
283where
284    F: MartianFileType,
285    W: Write,
286    T: Any + Serialize,
287{
288    writer: W,
289    phantom_f: PhantomData<F>,
290    phantom_t: PhantomData<T>,
291    processed_items: usize,
292}
293
294impl<T, F, W> LazyWrite<T, W> for LazyBincodeWriter<T, F, W>
295where
296    F: MartianFileType,
297    W: Write,
298    T: Any + Serialize,
299{
300    type FileType = BinaryFormat<F, Vec<T>>;
301    fn with_writer(mut writer: W) -> Result<Self, Error> {
302        let type_hash = type_name::<LazyMarker<Vec<T>>>(); // The file stores Vec<T>, not T
303        bincode::serialize_into(&mut writer, &type_hash)?;
304        Ok(LazyBincodeWriter {
305            writer,
306            phantom_f: PhantomData,
307            phantom_t: PhantomData,
308            processed_items: 0,
309        })
310    }
311    fn write_item(&mut self, item: &T) -> Result<(), Error> {
312        self.processed_items += 1;
313        bincode::serialize_into(&mut self.writer, &item)?;
314        Ok(())
315    }
316    fn finish(self) -> Result<W, Error> {
317        Ok(self.writer)
318    }
319}
